function adaptiveViewportWheelZoom(e){
  if(e.ctrlKey)return; // leave browser/system Ctrl+wheel behavior alone
  e.preventDefault();
  e.stopPropagation();

  const delta=Math.sign(e.deltaY);
  if(!delta)return;

  const toTarget=new THREE.Vector3().subVectors(orbit.target,camera.position);
  let distance=toTarget.length();
  if(distance<1e-6){
    camera.getWorldDirection(toTarget);
    distance=VIEW_ZOOM.closeDistance;
  }else{
    toTarget.multiplyScalar(1/distance);
  }

  // Normalize trackpads/mice without allowing a single event to jump excessively.
  const wheelUnits=Math.min(3,Math.max(1,Math.abs(e.deltaY)/100));
  const proportional=Math.min(distance*VIEW_ZOOM.maxFraction,
                              Math.max(distance*VIEW_ZOOM.fraction,VIEW_ZOOM.minStep));
  let step=proportional*wheelUnits;

  if(delta<0){ // zoom in
    if(distance>VIEW_ZOOM.closeDistance){
      // Approach the orbit target normally, but never collapse onto it.
      step=Math.min(step,Math.max(VIEW_ZOOM.minStep,distance-VIEW_ZOOM.closeDistance));
      camera.position.addScaledVector(toTarget,step);
    }else{
      // Near the target, translate camera and target together. This prevents
      // the familiar "zoom barely moves near the object" stall.
      camera.position.addScaledVector(toTarget,step);
      orbit.target.addScaledVector(toTarget,step);
    }
  }else{ // zoom out
    camera.position.addScaledVector(toTarget,-step);
  }

  orbit.update();
}

const RenderLightPolicy=Object.freeze({
  supported:Object.freeze(['point','spot','rectangle','disc','sphere','sun','dome','mesh']),
  minLinearSize:0.01,
  minArea:1e-4,
  finite3(a,fallback=[0,0,0]){
    if(!Array.isArray(a)||a.length<3||a.slice(0,3).some(v=>!Number.isFinite(Number(v))))return fallback.slice();
    return a.slice(0,3).map(Number);
  },
  type(t){return this.supported.includes(t)?t:'point';},
  scalar(v,fallback=0,min=-Infinity,max=Infinity){
    v=Number(v);if(!Number.isFinite(v))v=fallback;
    return Math.max(min,Math.min(max,v));
  },
  area(light){
    if(light.type==='rectangle'||light.type==='mesh')return Math.max(this.minArea,light.width*light.height);
    if(light.type==='disc')return Math.max(this.minArea,Math.PI*Math.pow(light.size*.5,2));
    if(light.type==='sphere')return Math.max(this.minArea,4*Math.PI*Math.pow(light.size*.5,2));

  }
