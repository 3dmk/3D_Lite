const RenderAccelerationCache16={
  geometryRevision:null,
  acceleration:null,
  hits:0,
  misses:0,
  reset(){this.geometryRevision=null;this.acceleration=null;this.hits=0;this.misses=0;},
  invalidate(){this.geometryRevision=null;this.acceleration=null;},
  revision(scene){
    const m=scene?.metadata||{};
    return [m.sceneRevision??'s?',m.geometryRevision??'g?',m.materialRevision??'m?',m.lightRevision??'l?',scene?.objects?.length||0,
      ...(scene?.objects||[]).map(o=>`${o.id}:${o.visible!==false?1:0}:${o.mesh?.positions?.length||0}:${o.mesh?.indices?.length||0}`)
    ].join('|');
  },
  get(scene){
    const rev=this.revision(scene);
    if(this.acceleration&&this.geometryRevision===rev){this.hits++;return this.acceleration;}
    this.misses++;return null;
  },
  put(scene,acceleration){
    this.geometryRevision=this.revision(scene);
    this.acceleration=acceleration;
    return acceleration;
  },
  stats(){return Object.freeze({geometryRevision:this.geometryRevision,hits:this.hits,misses:this.misses,reused:!!this.acceleration});}
};

/*

  - object mesh geometry revisions feed BLAS-equivalent cached acceleration data;
  - scene/object transform revisions are intentionally kept outside this cache key
    so a future TLAS can update instances without rebuilding object geometry.
  Current browser renderer still uses the proven binary RenderAccelerationSet.
*/
const RenderAccelerationHierarchy16=Object.freeze({
  mode:'geometry-revision acceleration cache; TLAS/BLAS boundary reserved',
  cache:RenderAccelerationCache16,
  build(scene,builder){
    const cached=RenderAccelerationCache16.get(scene);
    if(cached)return cached;
    return RenderAccelerationCache16.put(scene,builder());
  }
});
