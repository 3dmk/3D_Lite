  constructor(width,height){

  }
  add(index,color){
    const s=++this.samples[index];
    const o=index*3;
    this.sum[o]+=color[0];this.sum[o+1]+=color[1];this.sum[o+2]+=color[2];
    const lum=.2126*color[0]+.7152*color[1]+.0722*color[2];

  }

  }

    for(let i=0;i<this.width*this.height;i++){
