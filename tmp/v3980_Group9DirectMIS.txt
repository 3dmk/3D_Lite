const Group9DirectMIS=Object.freeze({
  lightPdf(light,sample){
    if(!light||!sample)return 0;
    if(['point','spot','sun'].includes(light.type))return Infinity; // delta lights: light sampling owns the event
    if(['rectangle','disc','sphere','mesh'].includes(light.type)){
      const area=Math.max(RenderLightPolicy.minArea,Number(sample.area)||RenderLightPolicy.area(light));
      const distance=Math.max(1e-6,Number(sample.distance)||1);
      const cosEmitter=Math.max(1e-6,Number(sample.emitterCos)||1);
      return (distance*distance)/(area*cosEmitter);
    }
    return 0;
  },

  evaluate(job,acceleration,compiled,ray,hit,pixelSeed=1){
    const material=RenderMaterialLibrary.resolveHit(job.renderScene,compiled.materials,hit);
    const n=SurfaceInterpolator.normal(acceleration,hit);
    const v=RayMath.mul(ray.direction,-1);
    let color=RenderColor.mul(material.emissive,material.emissiveIntensity);

    if(material.shading==='unlit')return RenderColor.add(material.baseColor,color);

    const useTree=LightTreePolicy.enabled(job,compiled)&&job.lightTree;
    const selected=useTree?job.lightTree.sample(hit.position,pixelSeed^0x51ED270B):null;
    const candidates=selected
      ? [{light:selected.light,sourceIndex:selected.sourceIndex,selectionPdf:selected.pdf}]
      : compiled.lights.map((light,sourceIndex)=>({light,sourceIndex,selectionPdf:1}));

    for(const candidate of candidates){
      const light=candidate.light;
      if(!light||light.type==='dome')continue;

      const li=candidate.sourceIndex;
      const selectionPdf=Math.max(1e-8,candidate.selectionPdf||1);
      const count=DirectLightingIntegrator.samplesFor(light,job.settings);
      let accum=[0,0,0];

      for(let si=0;si<count;si++){
        const uv=RenderSampling.sample2D(pixelSeed^(li*2654435761),si);
        const sample=['rectangle','disc','sphere','mesh'].includes(light.type)
          ? AreaLightSampler.sample(light,hit.position,uv[0],uv[1])
          : DirectLightSampler.sample(light,hit.position);

        if(!sample?.direction||!sample.radiance||sample.radiance.every(x=>x<=0||!Number.isFinite(x)))continue;
        if(light.castShadows!==false&&!ShadowVisibility.visible(acceleration,hit.position,sample.direction,sample.distance))continue;

        const scaledRadiance=useTree?RenderColor.mul(sample.radiance,1/selectionPdf):sample.radiance;
        const brdf=PhysicalBSDF.evaluate(material,n,v,sample.direction);
        const bsdfPdf=PathBSDFSampler.pdf(material,n,v,sample.direction);
        const conditionalPdf=this.lightPdf(light,sample);
        const lightPdf=Number.isFinite(conditionalPdf)?conditionalPdf*selectionPdf:conditionalPdf;
        const weight=Number.isFinite(lightPdf)?RenderMIS.power(lightPdf,bsdfPdf):1;
        accum=RenderColor.add(accum,RenderColor.mul(RenderColor.mul(brdf,scaledRadiance),weight));
      }

      if(count>1)accum=RenderColor.mul(accum,1/count);
      color=RenderColor.add(color,accum);
    }
    return color;
  }
}
