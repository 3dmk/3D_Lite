var RenderBVH = class RenderBVH{
  constructor(store,maxLeafSize=4){
    this.store=store;
    this.maxLeafSize=Math.max(1,maxLeafSize|0);
    this.nodes=[];
    this.primitiveIndices=Uint32Array.from({length:store.triangleCount()},(_,i)=>i);
    this._boundsCache=Array.from({length:store.triangleCount()},(_,i)=>store.triangleBounds(i));
    if(store.triangleCount())this.root=this._build(0,store.triangleCount());
    else this.root=-1;
  }
  _rangeBounds(start,end){
    let mn=[Infinity,Infinity,Infinity],mx=[-Infinity,-Infinity,-Infinity],cmn=[Infinity,Infinity,Infinity],cmx=[-Infinity,-Infinity,-Infinity];
    for(let i=start;i<end;i++){
      const b=this._boundsCache[this.primitiveIndices[i]];
      mn=RayMath.min3(mn,b.min);mx=RayMath.max3(mx,b.max);
      cmn=RayMath.min3(cmn,b.centroid);cmx=RayMath.max3(cmx,b.centroid);
    }
    return {min:mn,max:mx,cmin:cmn,cmax:cmx};
  }
  _surfaceArea(min,max){
    const x=Math.max(0,max[0]-min[0]),y=Math.max(0,max[1]-min[1]),z=Math.max(0,max[2]-min[2]);
    return 2*(x*y+y*z+z*x);
  }
  _sahSplit(start,end,rb){
    const count=end-start;
    if(count<8)return null;
    let best=null,bestCost=Infinity;
    const binsN=8;
    for(let axis=0;axis<3;axis++){
      const lo=rb.cmin[axis],hi=rb.cmax[axis],span=hi-lo;
      if(!(span>1e-12))continue;
      const bins=Array.from({length:binsN},()=>({count:0,min:[Infinity,Infinity,Infinity],max:[-Infinity,-Infinity,-Infinity]}));
      for(let i=start;i<end;i++){
        const pi=this.primitiveIndices[i],b=this._boundsCache[pi];
        const bi=Math.min(binsN-1,Math.max(0,Math.floor(((b.centroid[axis]-lo)/span)*binsN)));
        const bn=bins[bi];bn.count++;bn.min=RayMath.min3(bn.min,b.min);bn.max=RayMath.max3(bn.max,b.max);
      }
      const leftC=[],rightC=[],leftB=[],rightB=[];
      let c=0,mn=[Infinity,Infinity,Infinity],mx=[-Infinity,-Infinity,-Infinity];
      for(let i=0;i<binsN;i++){c+=bins[i].count;if(bins[i].count){mn=RayMath.min3(mn,bins[i].min);mx=RayMath.max3(mx,bins[i].max);}leftC[i]=c;leftB[i]={min:mn.slice(),max:mx.slice()};}
      c=0;mn=[Infinity,Infinity,Infinity];mx=[-Infinity,-Infinity,-Infinity];
      for(let i=binsN-1;i>=0;i--){c+=bins[i].count;if(bins[i].count){mn=RayMath.min3(mn,bins[i].min);mx=RayMath.max3(mx,bins[i].max);}rightC[i]=c;rightB[i]={min:mn.slice(),max:mx.slice()};}
      for(let i=0;i<binsN-1;i++){
        if(!leftC[i]||!rightC[i+1])continue;
        const cost=this._surfaceArea(leftB[i].min,leftB[i].max)*leftC[i]+this._surfaceArea(rightB[i+1].min,rightB[i+1].max)*rightC[i+1];
        if(cost<bestCost){bestCost=cost;best={axis,threshold:lo+span*((i+1)/binsN)};}
      }
    }
    return best;
  }
  _build(start,end){
    const rb=this._rangeBounds(start,end);
    const nodeIndex=this.nodes.length;
    this.nodes.push(null);
    const count=end-start;
    if(count<=this.maxLeafSize){
      this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,-1,-1,start,count);
      return nodeIndex;
    }
    const ext=RayMath.sub(rb.cmax,rb.cmin);
    let axis=0;if(ext[1]>ext[axis])axis=1;if(ext[2]>ext[axis])axis=2;
    const sah=this._sahSplit(start,end,rb);
    if(sah)axis=sah.axis;
    const sorted=new Array(end-start);for(let si=0;si<sorted.length;si++)sorted[si]=this.primitiveIndices[start+si];
    sorted.sort((a,b)=>this._boundsCache[a].centroid[axis]-this._boundsCache[b].centroid[axis]);
    this.primitiveIndices.set(sorted,start);
    let mid=start+Math.floor(count/2);
    if(sah){
      let split=start;
      while(split<end&&this._boundsCache[this.primitiveIndices[split]].centroid[axis]<sah.threshold)split++;
      if(split>start&&split<end)mid=split;
    }
    if(mid===start||mid===end){
      this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,-1,-1,start,count);
      return nodeIndex;
    }
    const left=this._build(start,mid),right=this._build(mid,end);
    this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,left,right,0,0);
    return nodeIndex;
  }
  intersect(ray){
    const best=new RenderHit();
    if(this.root<0)return best;
    const stack=[this.root];stack.length=1; // traversal stack retained; avoids unsafe structural rewrite
    while(stack.length){
      const ni=stack.pop(),node=this.nodes[ni];
      if(!AABBIntersector.intersect(ray,node.min,node.max,best.t))continue;
      if(node.leaf){
        for(let i=node.start;i<node.start+node.count;i++){
          const tri=this.primitiveIndices[i],k=tri*3;
          const p0=this.store.vertex(this.store.triangles[k]);
          const p1=this.store.vertex(this.store.triangles[k+1]);
          const p2=this.store.vertex(this.store.triangles[k+2]);
          const before=best.t;
          TriangleIntersector.intersect(ray,p0,p1,p2,best);
          if(best.t<before){
            best.triangleIndex=tri;
            best.objectIndex=this.store.objectIndices[tri];
            best.materialIndex=this.store.materialIndices[tri];
          }
        }
      }else{
        const left=this.nodes[node.left],right=this.nodes[node.right];
        const lt=AABBIntersector.entry(ray,left.min,left.max,best.t);
        const rt=AABBIntersector.entry(ray,right.min,right.max,best.t);
        const lh=Number.isFinite(lt),rh=Number.isFinite(rt);
        if(lh&&rh){
          if(lt<=rt)stack.push(node.right,node.left);else stack.push(node.left,node.right);
        }else if(lh)stack.push(node.left);
        else if(rh)stack.push(node.right);
      }
    }
    return best;
  }
  stats(){
    let leafCount=0,maxDepth=0;
    if(this.root>=0){
      const stack=[[this.root,0]];
      while(stack.length){
        const [i,d]=stack.pop(),n=this.nodes[i];maxDepth=Math.max(maxDepth,d);
        if(n.leaf)leafCount++;else{stack.push([n.left,d+1]);stack.push([n.right,d+1]);}
      }
    }
    return Object.freeze({nodeCount:this.nodes.length,leafCount,maxDepth,triangleCount:this.store.triangleCount(),maxLeafSize:this.maxLeafSize,buildStrategy:'SAH-8bin-binary'});
  }
  anyHit(ray){
    if(this.root<0)return false;
    const stack=[this.root];
    const temp=new RenderHit();
    while(stack.length){
      const node=this.nodes[stack.pop()];
      if(!AABBIntersector.intersect(ray,node.min,node.max,ray.tMax))continue;
      if(node.leaf){
        for(let i=node.start;i<node.start+node.count;i++){
          const tri=this.primitiveIndices[i],k=tri*3;
          temp.hit=false;temp.t=Infinity;
          TriangleIntersector.intersect(
            ray,
            this.store.vertex(this.store.triangles[k]),
            this.store.vertex(this.store.triangles[k+1]),
            this.store.vertex(this.store.triangles[k+2]),
            temp
          );
          if(temp.hit)return true;
        }
      }else{
        stack.push(node.left,node.right);
      }
    }
    return false;
  }
}
