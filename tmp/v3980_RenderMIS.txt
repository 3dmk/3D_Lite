const AreaLightSampler=Object.freeze({
  sample(light,point,u=.5,v=.5){
    if(light.type==='rectangle'){
      const lp=LightBasis.pointOnRectangle(light,u,v);
      return this._local(light,point,lp,RenderLightPolicy.area(light),light.direction);
    }
    if(light.type==='disc'){
      const lp=LightBasis.pointOnDisc(light,u,v);
      return this._local(light,point,lp,RenderLightPolicy.area(light),light.direction);
    }
    if(light.type==='sphere'){
      const lp=LightBasis.pointOnSphere(light,u,v);
      const normal=RayMath.normalize(RayMath.sub(lp,light.position));
      return this._local(light,point,lp,RenderLightPolicy.area(light),normal);
    }
    if(light.type==='mesh'){
      // Part 4 keeps Mesh Light as a rectangular proxy.
      // True emissive mesh-triangle sampling remains Group 10.
      const lp=LightBasis.pointOnRectangle(light,u,v);
      return this._local(light,point,lp,RenderLightPolicy.area(light),light.direction);
    }
    return DirectLightSampler.sample(light,point);
  },

  _local(light,point,lightPoint,area,emitterNormal=null){
    const to=RayMath.sub(lightPoint,point);
    const dist=Math.max(RayMath.length(to),1e-6);
    const dir=RayMath.mul(to,1/dist);

    if(light.range>0&&dist>=light.range){
      return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0],lightPoint,area};
    }

    let attenuation=1/(dist*dist);
    if(light.range>0){
      const f=Math.max(0,1-dist/light.range);
      attenuation*=f*f;
    }

    let emitterCos=1;
    if(emitterNormal){
      emitterCos=Math.max(0,RayMath.dot(RayMath.normalize(emitterNormal),RayMath.mul(dir,-1)));
      if(emitterCos<=0){
        return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0],lightPoint,area,emitterCos:0};
      }
    }

    const safeArea=Math.max(RenderLightPolicy.minArea,area);
    return {
      kind:'local',
      direction:dir,
      distance:dist,
      radiance:RenderColor.mul(light.color,light.intensity*attenuation*emitterCos/safeArea),
      lightPoint,
      area:safeArea,
      emitterCos
    };
  }
});

var DirectLightingIntegrator=Object.freeze({
  samplesFor(light,settings){
    if(['rectangle','disc','sphere','mesh'].includes(light.type)){
      const preset=String(settings?.quality||'Preview');
      if(preset==='Ultra')return 16;
      if(preset==='High')return 8;
      if(preset==='Draft')return 1;
      return 4;
    }
    return 1;
  },

  evaluate(job,acceleration,compiled,ray,hit,pixelSeed=1,options={}){
    const material=RenderMaterialLibrary.resolveHit(job.renderScene,compiled.materials,hit);
    const n=SurfaceInterpolator.normal(acceleration,hit);
    const v=RayMath.mul(ray.direction,-1);
    const includeEmission=options.includeEmission!==false;
    const includeDome=options.includeDome!==false;
    let color=includeEmission?RenderColor.mul(material.emissive,material.emissiveIntensity):[0,0,0];

    if(material.shading==='unlit'){
      return includeEmission?RenderColor.add(material.baseColor,color):[0,0,0];
    }

    const emissiveSet=compiled.emissiveTriangles||job.emissiveTriangles||null;
    if(emissiveSet?.triangles?.length){
      const es=EmissiveTriangleSet.sample(emissiveSet,pixelSeed,0,hit.position);
      if(es&&ShadowVisibility.visible(acceleration,hit.position,es.direction,es.distance)){
        const brdf=PhysicalBSDF.evaluate(material,n,v,es.direction);
        const bsdfPdf=PathBSDFSampler.pdf(material,n,v,es.direction);
        const w=RenderMIS.power(es.pdf,bsdfPdf);
        color=RenderColor.add(color,RenderColor.mul(RenderColor.mul(brdf,es.radiance),w));
      }
    }

    for(let li=0;li<compiled.lights.length;li++){
      const light=compiled.lights[li];

      if(light.type==='dome'){
        if(includeDome){
          const ambient=EnvironmentSampler.radiance(light,n);
          const diffuse=RenderColor.mul(material.baseColor,(1-material.metalness)/Math.PI);
          color=RenderColor.add(color,RenderColor.mul(RenderColor.mul(diffuse,ambient),Math.PI));
        }
        continue;
      }

      const count=this.samplesFor(light,job.settings);
      let accum=[0,0,0];

      for(let si=0;si<count;si++){
        const uv=RenderSampling.sample2D(pixelSeed^(li*2654435761),si);
        const sample=['rectangle','disc','sphere','mesh'].includes(light.type)
          ? AreaLightSampler.sample(light,hit.position,uv[0],uv[1])
          : DirectLightSampler.sample(light,hit.position);

        if(!sample||!sample.direction)continue;
        if(!sample.radiance||sample.radiance.every(v=>v<=0||!Number.isFinite(v)))continue;

        if(light.castShadows!==false){
          const visible=ShadowVisibility.visible(acceleration,hit.position,sample.direction,sample.distance);
          if(!visible)continue;
        }

        const brdf=PhysicalBSDF.evaluate(material,n,v,sample.direction);
        accum=RenderColor.add(accum,RenderColor.mul(brdf,sample.radiance));
      }

      if(count>1)accum=RenderColor.mul(accum,1/count);
      color=RenderColor.add(color,accum);
    }

    return color;
  }
});

const GGXImportanceSampler=Object.freeze({
  sampleHalfVector(n,roughness,u1,u2){
    const a=Math.max(.045,roughness);
    const a2=a*a;
    const phi=2*Math.PI*u1;
    const cosTheta=Math.sqrt(Math.max(0,(1-u2)/(1+(a2-1)*u2)));
    const sinTheta=Math.sqrt(Math.max(0,1-cosTheta*cosTheta));
    const basis=LightBasis.fromDirection(n);
    return RayMath.normalize(
      RayMath.add(
        RayMath.add(RayMath.mul(basis.tangent,sinTheta*Math.cos(phi)),RayMath.mul(basis.bitangent,sinTheta*Math.sin(phi))),
        RayMath.mul(basis.n,cosTheta)
      )
    );
  },
  sampleVisibleHalfVector(n,v,roughness,u1,u2){
    const a=Math.max(.045,roughness);
    const basis=LightBasis.fromDirection(n);
    const vl=[RayMath.dot(v,basis.tangent),RayMath.dot(v,basis.bitangent),Math.max(1e-6,RayMath.dot(v,basis.n))];
    const vh=RayMath.normalize([a*vl[0],a*vl[1],vl[2]]);
    const lensq=vh[0]*vh[0]+vh[1]*vh[1];
    const t1=lensq>1e-12?[-vh[1]/Math.sqrt(lensq),vh[0]/Math.sqrt(lensq),0]:[1,0,0];
    const t2=RayMath.cross(vh,t1);
    const r=Math.sqrt(Math.max(0,u1)),phi=2*Math.PI*u2;
    const p1=r*Math.cos(phi);
    let p2=r*Math.sin(phi);
    const mix=.5*(1+vh[2]);
    p2=(1-mix)*Math.sqrt(Math.max(0,1-p1*p1))+mix*p2;
    const z=Math.sqrt(Math.max(0,1-p1*p1-p2*p2));
    const nh=RayMath.add(RayMath.add(RayMath.mul(t1,p1),RayMath.mul(t2,p2)),RayMath.mul(vh,z));
    const ne=RayMath.normalize([a*nh[0],a*nh[1],Math.max(0,nh[2])]);
    return RayMath.normalize(RayMath.add(RayMath.add(RayMath.mul(basis.tangent,ne[0]),RayMath.mul(basis.bitangent,ne[1])),RayMath.mul(basis.n,ne[2])));
  },
  reflect(i,n){return RayMath.sub(i,RayMath.mul(n,2*RayMath.dot(i,n)));},
  distribution(nDotH,roughness){
    const a=Math.max(.045,roughness),a2=a*a;
    const d=nDotH*nDotH*(a2-1)+1;
    return a2/(Math.PI*d*d+1e-8);
  }
});

const DielectricTransport=Object.freeze({
  fresnelDielectric(cosI,etaI,etaT){
    cosI=Math.max(-1,Math.min(1,cosI));
    let entering=cosI>0;
    if(!entering){const t=etaI;etaI=etaT;etaT=t;cosI=Math.abs(cosI);}
    const sinI=Math.sqrt(Math.max(0,1-cosI*cosI));
    const sinT=etaI/etaT*sinI;
    if(sinT>=1)return 1;
    const cosT=Math.sqrt(Math.max(0,1-sinT*sinT));
    const rs=((etaT*cosI)-(etaI*cosT))/((etaT*cosI)+(etaI*cosT));
    const rp=((etaI*cosI)-(etaT*cosT))/((etaI*cosI)+(etaT*cosT));
    return Math.max(0,Math.min(1,(rs*rs+rp*rp)*.5));
  },
  refract(i,n,eta){
    const cosi=Math.max(-1,Math.min(1,RayMath.dot(i,n)));
    const k=1-eta*eta*(1-cosi*cosi);
    if(k<0)return null;
    return RayMath.normalize(RayMath.sub(RayMath.mul(i,eta),RayMath.mul(n,eta*cosi+Math.sqrt(k))));
  },
  beerLambert(color,distance,attenuationDistance){
    if(!Number.isFinite(attenuationDistance)||attenuationDistance<=0||distance<=0)return [1,1,1];
    return color.map(c=>Math.pow(Math.max(1e-6,c),distance/attenuationDistance));
  }
});

var PathBSDFSampler=Object.freeze({
  pdf(material,n,v,l){
    const nDotL=Math.max(0,RayMath.dot(n,l));
    if(nDotL<=0)return 0;
    const diffusePdf=nDotL/Math.PI;
    const specProb=Math.max(.08,Math.min(.92,.25+.65*material.metalness+.2*(1-material.roughness)));
    const h=RayMath.normalize(RayMath.add(v,l));
    if(!h.every(Number.isFinite))return (1-specProb)*diffusePdf;
    const nDotH=Math.max(0,RayMath.dot(n,h));
    const vDotH=Math.max(1e-7,RayMath.dot(v,h));
    const D=GGXImportanceSampler.distribution(nDotH,material.roughness);
    const specPdf=(D*nDotH)/(4*vDotH+1e-8);
    return (1-specProb)*diffusePdf+specProb*specPdf;
  },

  sample(material,n,v,seed,bounce){
    if(material.shading==='unlit')return null;
    const transmission=Math.max(0,Math.min(1,material.transmission||0));
    const chooseT=PathSampler.scalar(seed,bounce,6);

    if(transmission>0&&chooseT<transmission){
      const incident=RayMath.mul(v,-1);
      const entering=RayMath.dot(incident,n)<0;
      const normal=entering?n:RayMath.mul(n,-1);
      const etaI=entering?1:material.ior,etaT=entering?material.ior:1;
      const eta=etaI/etaT;
      const cosI=Math.abs(RayMath.dot(incident,normal));
      const F=DielectricTransport.fresnelDielectric(cosI,etaI,etaT);
      const chooseF=PathSampler.scalar(seed,bounce,16);
      let direction,lobe,weight;
      if(chooseF<F){
        direction=GGXImportanceSampler.reflect(incident,normal);
        lobe='dielectric-reflect';
        weight=[1,1,1];
      }else{
        direction=DielectricTransport.refract(incident,normal,eta);
        if(!direction){
          direction=GGXImportanceSampler.reflect(incident,normal);
          lobe='dielectric-tir';
          weight=[1,1,1];
        }else{
          lobe='transmission';
          weight=material.attenuationColor.slice();
        }
      }
      return {direction:RayMath.normalize(direction),pdf:1,weight,lobe,transmission:true};
    }

    const specProb=Math.max(.08,Math.min(.92,.25+.65*material.metalness+.2*(1-material.roughness)));
    const choose=PathSampler.scalar(seed,bounce,7);
    let direction,lobe='diffuse';

    if(choose<specProb){
      const u1=PathSampler.scalar(seed,bounce,8),u2=PathSampler.scalar(seed,bounce,9);
      const h=GGXImportanceSampler.sampleVisibleHalfVector(n,v,material.roughness,u1,u2);
      direction=GGXImportanceSampler.reflect(RayMath.mul(v,-1),h);
      direction=RayMath.normalize(direction);
      lobe='ggx';
      if(RayMath.dot(n,direction)<=0)direction=PathSampler.cosineHemisphere(n,seed^(0xA511E9B3>>>0),bounce);
    }else{
      direction=PathSampler.cosineHemisphere(n,seed,bounce);
    }

    const cosTheta=Math.max(0,RayMath.dot(n,direction));
    if(cosTheta<=0)return null;
    const value=PhysicalBSDF.evaluate(material,n,v,direction);
    const pdf=Math.max(this.pdf(material,n,v,direction),1e-8);
    return {direction,pdf,weight:RenderColor.mul(value,1/pdf),lobe};
  }
});

const Group9DirectMIS=Object.freeze({
  lightPdf(light,sample){
    if(!light||!sample)return 0;
    if(['point','spot','sun'].includes(light.type))return Infinity; // delta lights: light sampling owns the event
    if(['rectangle','disc','sphere','mesh'].includes(light.type)){
      const area=Math.max(RenderLightPolicy.minArea,Number(sample.area)||RenderLightPolicy.area(light));
      const distance=Math.max(1e-6,Number(sample.distance)||1);
      const cosEmitter=Math.max(1e-6,Number(sample.emitterCos)||1);
      return (distance*distance)/(area*cosEmitter);
    }
    return 0;
  },

  evaluate(job,acceleration,compiled,ray,hit,pixelSeed=1){
    const material=RenderMaterialLibrary.resolveHit(job.renderScene,compiled.materials,hit);
    const n=SurfaceInterpolator.normal(acceleration,hit);
    const v=RayMath.mul(ray.direction,-1);
    let color=RenderColor.mul(material.emissive,material.emissiveIntensity);
    const maxShadowRays=Math.max(1,Math.min(4096,Number(job?.settings?.maxShadowRaysPerHit)||1024));
    let shadowRays=0;

    if(material.shading==='unlit')return RenderColor.add(material.baseColor,color);

    const useTree=LightTreePolicy.enabled(job,compiled)&&job.lightTree;
    const selected=useTree?job.lightTree.sample(hit.position,pixelSeed^0x51ED270B):null;
    const candidates=selected
      ? [{light:selected.light,sourceIndex:selected.sourceIndex,selectionPdf:selected.pdf}]
      : compiled.lights.map((light,sourceIndex)=>({light,sourceIndex,selectionPdf:1}));

    for(const candidate of candidates){
      const light=candidate.light;
      if(!light||light.type==='dome')continue;

      const li=candidate.sourceIndex;
      const selectionPdf=Math.max(1e-8,candidate.selectionPdf||1);
      const count=DirectLightingIntegrator.samplesFor(light,job.settings);
      let accum=[0,0,0];

      for(let si=0;si<count;si++){
        const uv=RenderSampling.sample2D(pixelSeed^(li*2654435761),si);
        const sample=['rectangle','disc','sphere','mesh'].includes(light.type)
          ? AreaLightSampler.sample(light,hit.position,uv[0],uv[1])
          : DirectLightSampler.sample(light,hit.position);

        if(!sample?.direction||!sample.radiance||sample.radiance.every(x=>x<=0||!Number.isFinite(x)))continue;
        if(light.castShadows!==false){
          if(shadowRays>=maxShadowRays)break;
          shadowRays++;

        }

        const scaledRadiance=useTree?RenderColor.mul(sample.radiance,1/selectionPdf):sample.radiance;
        const brdf=PhysicalBSDF.evaluate(material,n,v,sample.direction);
        const bsdfPdf=PathBSDFSampler.pdf(material,n,v,sample.direction);
        const conditionalPdf=this.lightPdf(light,sample);
        const lightPdf=Number.isFinite(conditionalPdf)?conditionalPdf*selectionPdf:conditionalPdf;
        const weight=Number.isFinite(lightPdf)?RenderMIS.power(lightPdf,bsdfPdf):1;
        accum=RenderColor.add(accum,RenderColor.mul(RenderColor.mul(brdf,scaledRadiance),weight));
      }

      if(count>1)accum=RenderColor.mul(accum,1/count);
      color=RenderColor.add(color,accum);
    }
    return color;
  }
});

        if(d<this.radius&&d<bestD&&RayMath.dot(r.normal,normal)>=LightCachePolicy.normalThreshold){best=id;bestD=d;}

const LightCacheEstimator=Object.freeze({
  enabled(job){return job?.settings?.globalIllumination!==false&&job?.settings?.secondaryGI==='Light Cache';},
  cacheable(material){return material&&material.shading!=='unlit'&&(material.transmission||0)<.05&&(material.metalness||0)<.95;},

  probe(job,acceleration,compiled,hit,n,material,seed,bounce){
    const count=LightCachePolicy.probes(job);
    let incoming=[0,0,0];
    for(let i=0;i<count;i++){

      const origin=PathIntegrator.offsetOrigin(hit,n,d);

      try{LiteTraceExecutionContext.performance?.addProbe?.();}catch(_){}

      }
      incoming=RenderColor.add(incoming,li);
    }
    incoming=RenderColor.mul(incoming,1/Math.max(1,count));
    const diffuse=RenderColor.mul(material.baseColor,Math.max(0,1-material.metalness));
    return RenderColor.mul(diffuse,incoming);
  },

  estimate(job,acceleration,compiled,hit,n,material,seed,bounce){
    const cache=job.lightCache;if(!cache||!this.cacheable(material))return null;
    const found=cache.query(hit.position,n);
    if(found){
      const cadence=LightCachePolicy.retraceEvery(job);
      if(cadence>0&&cache.hits%cadence===0){
        cache.retraces++;
        const fresh=this.probe(job,acceleration,compiled,hit,n,material,seed^0xB5297A4D,bounce);
        cache.refreshNearest(hit.position,n,fresh);
        return RenderColor.mul(RenderColor.add(RenderColor.mul(found,.75),RenderColor.mul(fresh,.25)),1);
      }

    }

  }

var PathIntegrator=Object.freeze({
  maxDepth(job){
    return Math.max(1,Math.min(16,Number(job?.settings?.maxBounces)||5));
  },

  finiteColor(c){
    return Array.isArray(c)&&c.length===3&&c.every(v=>Number.isFinite(v)&&v>=0);
  },

  safeColor(c){
    return this.finiteColor(c)?c:[0,0,0];
  },

  offsetOrigin(hit,n,direction){
    const sign=RayMath.dot(direction,n)>=0?1:-1;
    const eps=RenderRayPolicy.epsilonForPoint(hit.position);
    return RayMath.add(hit.position,RayMath.mul(n,eps*sign));
  },

  trace(job,acceleration,compiled,primaryRay,pixelSeed,initialHit=null){
    let radiance=[0,0,0];
    let throughput=[1,1,1];
    let ray=primaryRay;
    let previousGuide=null;
    const maxDepth=this.maxDepth(job);
    const maxPathRays=Math.max(1,Math.min(4096,Number(job?.settings?.maxPathRays)||64));
    let pathRays=1;

    for(let bounce=0;bounce<maxDepth;bounce++){
      const hit=(bounce===0&&initialHit)?initialHit:acceleration.trace(ray);

      if(!hit.hit){
        const env=this.safeColor(RenderCoreD.evaluateEnvironment(job,compiled,ray.direction));
        radiance=RenderColor.add(radiance,RenderColor.mul(throughput,env));
        if(previousGuide&&job.pathGuide){
          const w=Math.max(0,.2126*env[0]+.7152*env[1]+.0722*env[2]);
          job.pathGuide.record(previousGuide.position,previousGuide.normal,previousGuide.direction,w);
        }
        break;
      }

      const material=RenderMaterialLibrary.resolveHit(job.renderScene,compiled.materials,hit);
      const n=SurfaceInterpolator.normal(acceleration,hit);
      const v=RayMath.mul(ray.direction,-1);

      // Next-event direct lighting at each hit. Dome is excluded here because
      // environment energy is accounted for by path miss evaluation.
      const direct=Group9DirectMIS.evaluate(
        job,acceleration,compiled,ray,hit,pixelSeed^(bounce*0x9E3779B9)
      );
      const safeDirect=this.safeColor(direct);
      radiance=RenderColor.add(radiance,RenderColor.mul(throughput,safeDirect));

      if(previousGuide&&job.pathGuide){
        const w=Math.max(0,.2126*safeDirect[0]+.7152*safeDirect[1]+.0722*safeDirect[2]);
        job.pathGuide.record(previousGuide.position,previousGuide.normal,previousGuide.direction,w);
      }

      if(bounce>=1){
        if(job?.settings?.secondaryGI==='None')break;
        if(LightCacheEstimator.enabled(job)){
          const cachedTail=LightCacheEstimator.estimate(job,acceleration,compiled,hit,n,material,pixelSeed,bounce);
          if(cachedTail){
            radiance=RenderColor.add(radiance,RenderColor.mul(throughput,cachedTail));
            break;
          }
        }
      }

      if(material.shading==='unlit')break;

      const sample=GuidedBSDFSampler.sample(job,material,n,v,hit.position,pixelSeed,bounce);
      if(!sample)break;

      throughput=RenderColor.mul(throughput,sample.weight);

      if(!this.finiteColor(throughput))break;
      if(Math.max(throughput[0],throughput[1],throughput[2])<1e-6)break;

      // Russian roulette after the first few bounces. Survival probability tracks path throughput.
      if(bounce>=2){
        const survive=Math.max(.08,Math.min(.95,Math.max(throughput[0],throughput[1],throughput[2])));
        if(PathSampler.scalar(pixelSeed,bounce,15)>survive)break;
        throughput=RenderColor.mul(throughput,1/survive);
      }

      if(sample.transmission&&material.thickness>0){
        throughput=RenderColor.mul(
          throughput,
          DielectricTransport.beerLambert(material.attenuationColor,material.thickness,material.attenuationDistance)
        );
      }

      previousGuide=(!sample.transmission&&job.pathGuide&&RayMath.dot(n,sample.direction)>0)
        ? {position:hit.position.slice(),normal:n.slice(),direction:sample.direction.slice()}
        : null;

      if(pathRays>=maxPathRays)break;
      pathRays++;
      const origin=this.offsetOrigin(hit,n,sample.direction);
      const eps=RenderRayPolicy.epsilonForPoint(origin);
      const rayType=sample.transmission?RenderRayType.TRANSMISSION:(sample.lobe==='ggx'?RenderRayType.GLOSSY:RenderRayType.DIFFUSE);
      RenderCoreH.attachPerformance(job).addRay(rayType);
      ray=new RenderRay(origin,sample.direction,eps,Infinity,rayType,bounce+1);
    }

    return radiance;
  },

  tracePixel(job,acceleration,compiled,primaryRay,pixelSeed){
    const primaryHitRecord=acceleration.trace(primaryRay);
    if(!primaryHitRecord.hit){
      const env=this.safeColor(RenderCoreD.evaluateEnvironment(job,compiled,primaryRay.direction));
      return {radiance:env,primaryHit:false,primaryHitRecord:null};
    }
    return {
      radiance:this.trace(job,acceleration,compiled,primaryRay,pixelSeed,primaryHitRecord),
      primaryHit:true,
      primaryHitRecord
    };
  }
});
