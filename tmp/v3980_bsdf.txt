const PhysicalBSDF=Object.freeze({
  fresnelSchlick(cosTheta,f0){
    const m=Math.pow(1-Math.max(0,Math.min(1,cosTheta)),5);
    return [
      f0[0]+(1-f0[0])*m,
      f0[1]+(1-f0[1])*m,
      f0[2]+(1-f0[2])*m
    ];
  },
  distributionGGX(nDotH,roughness){
    const a=roughness*roughness,a2=a*a;
    const d=nDotH*nDotH*(a2-1)+1;
    return a2/Math.max(Math.PI*d*d,1e-7);
  },
  geometrySchlickGGX(nDotV,roughness){
    const r=roughness+1,k=(r*r)/8;
    return nDotV/Math.max(nDotV*(1-k)+k,1e-7);
  },
  geometrySmith(nDotV,nDotL,roughness){
    return this.geometrySchlickGGX(nDotV,roughness)*this.geometrySchlickGGX(nDotL,roughness);
  },
  evaluate(material,n,v,l){
    if(material.shading==='unlit')return material.baseColor;
    if(!material||![...n,...v,...l].every(Number.isFinite))return [0,0,0];

    const h=RayMath.normalize(RayMath.add(v,l));
    if(!h.every(Number.isFinite))return [0,0,0];
    const nDotL=Math.max(0,RayMath.dot(n,l));
    const nDotV=Math.max(0,RayMath.dot(n,v));
    const nDotH=Math.max(0,RayMath.dot(n,h));
    const vDotH=Math.max(0,RayMath.dot(v,h));
    if(nDotL<=0||nDotV<=0)return [0,0,0];

    const dielectricF0=[.04,.04,.04];
    const f0=RenderColor.mix(dielectricF0,material.baseColor,material.metalness);
    const F=this.fresnelSchlick(vDotH,f0);
    const D=this.distributionGGX(nDotH,material.roughness);
    const G=this.geometrySmith(nDotV,nDotL,material.roughness);

    const specScale=D*G/Math.max(4*nDotV*nDotL,1e-6);
    const spec=RenderColor.mul(F,specScale);

    const kD=[
      (1-F[0])*(1-material.metalness),
      (1-F[1])*(1-material.metalness),
      (1-F[2])*(1-material.metalness)
    ];
    const diffuse=[
      kD[0]*material.baseColor[0]/Math.PI,
      kD[1]*material.baseColor[1]/Math.PI,
      kD[2]*material.baseColor[2]/Math.PI
    ];
    const out=RenderColor.mul(RenderColor.add(diffuse,spec),nDotL);
    return out.map(x=>Number.isFinite(x)&&x>=0?x:0);
  }
});

const SurfaceInterpolator=Object.freeze({
  normal(acceleration,hit){
    const s=acceleration.store,t=hit.triangleIndex,k=t*3;
    if(!s.normals||s.normals.length!==s.positions.length)return hit.geometricNormal;
    const ia=s.triangles[k],ib=s.triangles[k+1],ic=s.triangles[k+2];
    const w=1-hit.u-hit.v;
    const read=i=>[s.normals[i*3],s.normals[i*3+1],s.normals[i*3+2]];
    const a=read(ia),b=read(ib),c=read(ic);
    let n=RayMath.normalize([
      a[0]*w+b[0]*hit.u+c[0]*hit.v,
      a[1]*w+b[1]*hit.u+c[1]*hit.v,
      a[2]*w+b[2]*hit.u+c[2]*hit.v
    ]);
    if(!n.every(Number.isFinite)||RayMath.length(n)<1e-10)n=hit.geometricNormal;
    if(RayMath.dot(n,hit.geometricNormal)<0)n=RayMath.mul(n,-1);
    return n;
  },
  uv(acceleration,hit){
    const s=acceleration.store,t=hit.triangleIndex,k=t*3;
    if(!s.uvs||s.uvs.length!==(s.positions.length/3)*2)return [0,0];
    const ia=s.triangles[k],ib=s.triangles[k+1],ic=s.triangles[k+2],w=1-hit.u-hit.v;
    return [
      s.uvs[ia*2]*w+s.uvs[ib*2]*hit.u+s.uvs[ic*2]*hit.v,
      s.uvs[ia*2+1]*w+s.uvs[ib*2+1]*hit.u+s.uvs[ic*2+1]*hit.v
    ];
  }
});

const RenderLightPolicy=Object.freeze({
  supported:Object.freeze(['point','spot','rectangle','disc','sphere','sun','dome','mesh']),
  minLinearSize:0.01,
  minArea:1e-4,
  finite3(a,fallback=[0,0,0]){
    if(!Array.isArray(a)||a.length<3||a.slice(0,3).some(v=>!Number.isFinite(Number(v))))return fallback.slice();
    return a.slice(0,3).map(Number);
  },
  type(t){return this.supported.includes(t)?t:'point';},
  scalar(v,fallback=0,min=-Infinity,max=Infinity){
    v=Number(v);if(!Number.isFinite(v))v=fallback;
    return Math.max(min,Math.min(max,v));
  },
  area(light){
    if(light.type==='rectangle'||light.type==='mesh')return Math.max(this.minArea,light.width*light.height);
    if(light.type==='disc')return Math.max(this.minArea,Math.PI*Math.pow(light.size*.5,2));
    if(light.type==='sphere')return Math.max(this.minArea,4*Math.PI*Math.pow(light.size*.5,2));
    return 1;
  }
});

const DirectLightSampler=Object.freeze({
  sample(light,point){
    if(light.type==='dome'){
      return {kind:'ambient',direction:null,distance:Infinity,radiance:RenderColor.mul(light.color,light.intensity)};
    }
    if(light.type==='sun'){
      return {kind:'directional',direction:RayMath.mul(light.direction,-1),distance:Infinity,radiance:RenderColor.mul(light.color,light.intensity)};
    }

    const to=RayMath.sub(light.position,point);
    const dist=Math.max(RayMath.length(to),1e-4);
    const dir=RayMath.mul(to,1/dist);
    let attenuation=1/(dist*dist);

    if(light.range>0){
      if(dist>=light.range)return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0]};
      const f=Math.max(0,1-dist/light.range);
      attenuation*=f*f;
    }

    if(light.type==='spot'){
      const fromLight=RayMath.mul(dir,-1);
      const cosTheta=RayMath.dot(light.direction,fromLight);
      const outer=Math.cos(light.coneAngle*.5);
      const inner=Math.cos(light.coneAngle*.5*(1-light.penumbra));
      const cone=inner<=outer?(cosTheta>=outer?1:0):Math.max(0,Math.min(1,(cosTheta-outer)/(inner-outer)));
      if(cone<=0)return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0]};
      attenuation*=cone;
    }

    // Area emitters use center-point evaluation in Group 6.
    // Group 7 replaces this with actual surface sampling + visibility.
    const areaScale=
      light.type==='rectangle'?Math.max(.01,light.width*light.height):
      light.type==='disc'?Math.max(.01,Math.PI*light.size*light.size*.25):
      light.type==='sphere'?Math.max(.01,4*Math.PI*Math.pow(light.size*.5,2)):
      light.type==='mesh'?Math.max(.01,light.width*light.height):
      1;

    if(['rectangle','disc','sphere','mesh'].includes(light.type))attenuation*=Math.sqrt(areaScale);

    return {
      kind:'local',
      direction:dir,
      distance:dist,
      radiance:RenderColor.mul(light.color,light.intensity*attenuation)
    };
  }
});

const RenderLightPolicy=Object.freeze({
  supported:Object.freeze(['point','spot','rectangle','disc','sphere','sun','dome','mesh']),
  minLinearSize:0.01,
  minArea:1e-4,
  finite3(a,fallback=[0,0,0]){
    if(!Array.isArray(a)||a.length<3||a.slice(0,3).some(v=>!Number.isFinite(Number(v))))return fallback.slice();
    return a.slice(0,3).map(Number);
  },
  type(t){return this.supported.includes(t)?t:'point';},
  scalar(v,fallback=0,min=-Infinity,max=Infinity){
    v=Number(v);if(!Number.isFinite(v))v=fallback;
    return Math.max(min,Math.min(max,v));
  },
  area(light){
    if(light.type==='rectangle'||light.type==='mesh')return Math.max(this.minArea,light.width*light.height);
    if(light.type==='disc')return Math.max(this.minArea,Math.PI*Math.pow(light.size*.5,2));
    if(light.type==='sphere')return Math.max(this.minArea,4*Math.PI*Math.pow(light.size*.5,2));
    return 1;
  }
});

class RenderSceneLight{
  constructor(snapshot){
    this.id=String(snapshot.id||'light');
    this.name=String(snapshot.name||'Light');
    this.type=RenderLightPolicy.type(snapshot.type);
    this.enabled=snapshot.enabled!==false;
    this.color=Object.freeze(RenderColor.hexToLinear(snapshot.color||'#ffffff'));
    this.intensity=RenderLightPolicy.scalar(snapshot.intensity,0,0,1e12);
    this.range=RenderLightPolicy.scalar(snapshot.range,0,0,1e12);
    this.size=RenderLightPolicy.scalar(snapshot.size,1,RenderLightPolicy.minLinearSize,1e9);
    this.width=RenderLightPolicy.scalar(snapshot.width,1,RenderLightPolicy.minLinearSize,1e9);
    this.height=RenderLightPolicy.scalar(snapshot.height,1,RenderLightPolicy.minLinearSize,1e9);
    this.coneAngle=RenderLightPolicy.scalar(snapshot.coneAngle,45,1,179)*Math.PI/180;
    this.penumbra=RenderLightPolicy.scalar(snapshot.penumbra,0,0,1);
    this.castShadows=snapshot.castShadows!==false;
    this.position=Object.freeze(RenderLightPolicy.finite3(snapshot.position,[0,0,0]));

    Object.freeze(this);
  }

const DirectLightSampler=Object.freeze({
  sample(light,point){
    if(light.type==='dome'){
      return {kind:'ambient',direction:null,distance:Infinity,radiance:RenderColor.mul(light.color,light.intensity)};
    }
    if(light.type==='sun'){
      return {kind:'directional',direction:RayMath.mul(light.direction,-1),distance:Infinity,radiance:RenderColor.mul(light.color,light.intensity)};
    }

    const to=RayMath.sub(light.position,point);
    const dist=Math.max(RayMath.length(to),1e-4);
    const dir=RayMath.mul(to,1/dist);
    let attenuation=1/(dist*dist);

    if(light.range>0){
      if(dist>=light.range)return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0]};
      const f=Math.max(0,1-dist/light.range);
      attenuation*=f*f;
    }

    if(light.type==='spot'){
      const fromLight=RayMath.mul(dir,-1);
      const cosTheta=RayMath.dot(light.direction,fromLight);
      const outer=Math.cos(light.coneAngle*.5);
      const inner=Math.cos(light.coneAngle*.5*(1-light.penumbra));
      const cone=inner<=outer?(cosTheta>=outer?1:0):Math.max(0,Math.min(1,(cosTheta-outer)/(inner-outer)));
      if(cone<=0)return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0]};
      attenuation*=cone;
    }

    // Area emitters use center-point evaluation in Group 6.
    // Group 7 replaces this with actual surface sampling + visibility.
    const areaScale=
      light.type==='rectangle'?Math.max(.01,light.width*light.height):
      light.type==='disc'?Math.max(.01,Math.PI*light.size*light.size*.25):
      light.type==='sphere'?Math.max(.01,4*Math.PI*Math.pow(light.size*.5,2)):
      light.type==='mesh'?Math.max(.01,light.width*light.height):
      1;

    if(['rectangle','disc','sphere','mesh'].includes(light.type))attenuation*=Math.sqrt(areaScale);

    return {
      kind:'local',
      direction:dir,
      distance:dist,
      radiance:RenderColor.mul(light.color,light.intensity*attenuation)
    };
  }
});

const RenderLightPolicy=Object.freeze({
  supported:Object.freeze(['point','spot','rectangle','disc','sphere','sun','dome','mesh']),
  minLinearSize:0.01,
  minArea:1e-4,
  finite3(a,fallback=[0,0,0]){
    if(!Array.isArray(a)||a.length<3||a.slice(0,3).some(v=>!Number.isFinite(Number(v))))return fallback.slice();
    return a.slice(0,3).map(Number);
  },
  type(t){return this.supported.includes(t)?t:'point';},
  scalar(v,fallback=0,min=-Infinity,max=Infinity){
    v=Number(v);if(!Number.isFinite(v))v=fallback;
    return Math.max(min,Math.min(max,v));
  },
  area(light){
    if(light.type==='rectangle'||light.type==='mesh')return Math.max(this.minArea,light.width*light.height);
    if(light.type==='disc')return Math.max(this.minArea,Math.PI*Math.pow(light.size*.5,2));
    if(light.type==='sphere')return Math.max(this.minArea,4*Math.PI*Math.pow(light.size*.5,2));
    return 1;
  }
});

const DirectLightSampler=Object.freeze({
  sample(light,point){
    if(light.type==='dome'){
      return {kind:'ambient',direction:null,distance:Infinity,radiance:RenderColor.mul(light.color,light.intensity)};
    }
    if(light.type==='sun'){
      return {kind:'directional',direction:RayMath.mul(light.direction,-1),distance:Infinity,radiance:RenderColor.mul(light.color,light.intensity)};
    }

    const to=RayMath.sub(light.position,point);
    const dist=Math.max(RayMath.length(to),1e-4);
    const dir=RayMath.mul(to,1/dist);
    let attenuation=1/(dist*dist);

    if(light.range>0){
      if(dist>=light.range)return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0]};
      const f=Math.max(0,1-dist/light.range);
      attenuation*=f*f;
    }

    if(light.type==='spot'){
      const fromLight=RayMath.mul(dir,-1);
      const cosTheta=RayMath.dot(light.direction,fromLight);
      const outer=Math.cos(light.coneAngle*.5);
      const inner=Math.cos(light.coneAngle*.5*(1-light.penumbra));
      const cone=inner<=outer?(cosTheta>=outer?1:0):Math.max(0,Math.min(1,(cosTheta-outer)/(inner-outer)));
      if(cone<=0)return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0]};
      attenuation*=cone;
    }

    // Area emitters use center-point evaluation in Group 6.
    // Group 7 replaces this with actual surface sampling + visibility.
    const areaScale=
      light.type==='rectangle'?Math.max(.01,light.width*light.height):
      light.type==='disc'?Math.max(.01,Math.PI*light.size*light.size*.25):
      light.type==='sphere'?Math.max(.01,4*Math.PI*Math.pow(light.size*.5,2)):
      light.type==='mesh'?Math.max(.01,light.width*light.height):
      1;

    if(['rectangle','disc','sphere','mesh'].includes(light.type))attenuation*=Math.sqrt(areaScale);

    return {
      kind:'local',
      direction:dir,
      distance:dist,
      radiance:RenderColor.mul(light.color,light.intensity*attenuation)
    };
  }
});

const RenderSampling=Object.freeze({
  hash32(x){
    x=(x|0)+0x6D2B79F5|0;
    x=Math.imul(x^(x>>>15),x|1);
    x^=x+Math.imul(x^(x>>>7),x|61);
    return ((x^(x>>>14))>>>0)/4294967296;
  },
  sample2D(seed,index){
    return [
      this.hash32((seed*73856093)^(index*19349663)),
      this.hash32((seed*83492791)^(index*2971215073))
    ];
  },
  concentricDisk(u1,u2){
    const sx=2*u1-1,sy=2*u2-1;
    if(sx===0&&sy===0)return [0,0];
    let r,theta;
    if(Math.abs(sx)>Math.abs(sy)){
      r=sx;theta=(Math.PI/4)*(sy/sx);
    }else{
      r=sy;theta=(Math.PI/2)-(Math.PI/4)*(sx/sy);
    }
    return [r*Math.cos(theta),r*Math.sin(theta)];
  }
});

const LightBasis=Object.freeze({
  fromDirection(direction){
    const n=RayMath.normalize(direction);
    const helper=Math.abs(n[2])<.999?[0,0,1]:[0,1,0];
    const tangent=RayMath.normalize(RayMath.cross(helper,n));
    const bitangent=RayMath.cross(n,tangent);
    return {n,tangent,bitangent};
  },
  pointOnRectangle(light,u,v){
    const b=this.fromDirection(light.direction);
    const du=(u-.5)*light.width,dv=(v-.5)*light.height;
    return RayMath.add(light.position,RayMath.add(RayMath.mul(b.tangent,du),RayMath.mul(b.bitangent,dv)));
  },
  pointOnDisc(light,u,v){
    const b=this.fromDirection(light.direction);
    const d=RenderSampling.concentricDisk(u,v);
    const r=light.size*.5;
    return RayMath.add(light.position,RayMath.add(RayMath.mul(b.tangent,d[0]*r),RayMath.mul(b.bitangent,d[1]*r)));
  },
  pointOnSphere(light,u,v){
    const z=1-2*u;
    const r=Math.sqrt(Math.max(0,1-z*z));
    const phi=2*Math.PI*v;
    const local=[r*Math.cos(phi),r*Math.sin(phi),z];
    return RayMath.add(light.position,RayMath.mul(local,light.size*.5));
  }
});

  },

  }
});

const ShadowVisibility=Object.freeze({
  visible(acceleration,origin,direction,maxDistance){
    try{LiteTraceExecutionContext.performance?.addRay?.(RenderRayType.SHADOW);}catch(_){}
    const eps=RenderRayPolicy.epsilonForPoint(origin);
    if(Number.isFinite(maxDistance)&&maxDistance<=eps*2)return true;
    const o=RayMath.add(origin,RayMath.mul(direction,eps));
    const limit=Number.isFinite(maxDistance)?Math.max(eps,maxDistance-eps):Infinity;
    const ray=new RenderRay(o,direction,eps,limit,RenderRayType.SHADOW,0);
    return !acceleration.occluded(ray);
  }

const AreaLightSampler=Object.freeze({
  sample(light,point,u=.5,v=.5){
    if(light.type==='rectangle'){
      const lp=LightBasis.pointOnRectangle(light,u,v);
      return this._local(light,point,lp,RenderLightPolicy.area(light),light.direction);
    }
    if(light.type==='disc'){
      const lp=LightBasis.pointOnDisc(light,u,v);
      return this._local(light,point,lp,RenderLightPolicy.area(light),light.direction);
    }
    if(light.type==='sphere'){
      const lp=LightBasis.pointOnSphere(light,u,v);
      const normal=RayMath.normalize(RayMath.sub(lp,light.position));
      return this._local(light,point,lp,RenderLightPolicy.area(light),normal);
    }
    if(light.type==='mesh'){
      // Part 4 keeps Mesh Light as a rectangular proxy.
      // True emissive mesh-triangle sampling remains Group 10.
      const lp=LightBasis.pointOnRectangle(light,u,v);
      return this._local(light,point,lp,RenderLightPolicy.area(light),light.direction);
    }
    return DirectLightSampler.sample(light,point);
  },

  _local(light,point,lightPoint,area,emitterNormal=null){
    const to=RayMath.sub(lightPoint,point);
    const dist=Math.max(RayMath.length(to),1e-6);
    const dir=RayMath.mul(to,1/dist);

    if(light.range>0&&dist>=light.range){
      return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0],lightPoint,area};
    }

    let attenuation=1/(dist*dist);
    if(light.range>0){
      const f=Math.max(0,1-dist/light.range);
      attenuation*=f*f;
    }

    let emitterCos=1;
    if(emitterNormal){
      emitterCos=Math.max(0,RayMath.dot(RayMath.normalize(emitterNormal),RayMath.mul(dir,-1)));
      if(emitterCos<=0){
        return {kind:'local',direction:dir,distance:dist,radiance:[0,0,0],lightPoint,area,emitterCos:0};
      }
    }

    const safeArea=Math.max(RenderLightPolicy.minArea,area);
    return {
      kind:'local',
      direction:dir,
      distance:dist,
      radiance:RenderColor.mul(light.color,light.intensity*attenuation*emitterCos/safeArea),
      lightPoint,
      area:safeArea,
      emitterCos
    };
  }
});

var DirectLightingIntegrator=Object.freeze({
  samplesFor(light,settings){
    if(['rectangle','disc','sphere','mesh'].includes(light.type)){
      const preset=String(settings?.quality||'Preview');
      if(preset==='Ultra')return 16;
      if(preset==='High')return 8;
      if(preset==='Draft')return 1;
      return 4;
    }
    return 1;
  },

  evaluate(job,acceleration,compiled,ray,hit,pixelSeed=1,options={}){
    const material=RenderMaterialLibrary.resolveHit(job.renderScene,compiled.materials,hit);
    const n=SurfaceInterpolator.normal(acceleration,hit);
    const v=RayMath.mul(ray.direction,-1);
    const includeEmission=options.includeEmission!==false;
    const includeDome=options.includeDome!==false;
    let color=includeEmission?RenderColor.mul(material.emissive,material.emissiveIntensity):[0,0,0];

    if(material.shading==='unlit'){
      return includeEmission?RenderColor.add(material.baseColor,color):[0,0,0];
    }

    const emissiveSet=compiled.emissiveTriangles||job.emissiveTriangles||null;
    if(emissiveSet?.triangles?.length){
      const es=EmissiveTriangleSet.sample(emissiveSet,pixelSeed,0,hit.position);
      if(es&&ShadowVisibility.visible(acceleration,hit.position,es.direction,es.distance)){
        const brdf=PhysicalBSDF.evaluate(material,n,v,es.direction);
        const bsdfPdf=PathBSDFSampler.pdf(material,n,v,es.direction);
        const w=RenderMIS.power(es.pdf,bsdfPdf);
        color=RenderColor.add(color,RenderColor.mul(RenderColor.mul(brdf,es.radiance),w));
      }
    }

    for(let li=0;li<compiled.lights.length;li++){
      const light=compiled.lights[li];

      if(light.type==='dome'){
        if(includeDome){
          const ambient=EnvironmentSampler.radiance(light,n);
          const diffuse=RenderColor.mul(material.baseColor,(1-material.metalness)/Math.PI);
          color=RenderColor.add(color,RenderColor.mul(RenderColor.mul(diffuse,ambient),Math.PI));
        }
        continue;
      }

      const count=this.samplesFor(light,job.settings);
      let accum=[0,0,0];

      for(let si=0;si<count;si++){
        const uv=RenderSampling.sample2D(pixelSeed^(li*2654435761),si);
        const sample=['rectangle','disc','sphere','mesh'].includes(light.type)
          ? AreaLightSampler.sample(light,hit.position,uv[0],uv[1])
          : DirectLightSampler.sample(light,hit.position);

        if(!sample||!sample.direction)continue;
        if(!sample.radiance||sample.radiance.every(v=>v<=0||!Number.isFinite(v)))continue;

        if(light.castShadows!==false){
          const visible=ShadowVisibility.visible(acceleration,hit.position,sample.direction,sample.distance);
          if(!visible)continue;
        }

        const brdf=PhysicalBSDF.evaluate(material,n,v,sample.direction);
        accum=RenderColor.add(accum,RenderColor.mul(brdf,sample.radiance));
      }

      if(count>1)accum=RenderColor.mul(accum,1/count);
      color=RenderColor.add(color,accum);
    }

    return color;
  }
});

const PathSampler=Object.freeze({
  scalar(seed,bounce,dimension){
    return RenderSampling.hash32(
      ((seed>>>0) ^ Math.imul((bounce+1),0x9E3779B1) ^ Math.imul((dimension+1),0x85EBCA77))>>>0
    );
  },

  cosineHemisphere(n,seed,bounce){
    const u1=this.scalar(seed,bounce,0);
    const u2=this.scalar(seed,bounce,1);
    const r=Math.sqrt(u1);
    const phi=2*Math.PI*u2;
    const x=r*Math.cos(phi),y=r*Math.sin(phi),z=Math.sqrt(Math.max(0,1-u1));

    const basis=LightBasis.fromDirection(n);
    return RayMath.normalize(
      RayMath.add(
        RayMath.add(RayMath.mul(basis.tangent,x),RayMath.mul(basis.bitangent,y)),
        RayMath.mul(basis.n,z)
      )
    );
  }
});

var RenderMIS=Object.freeze({
  power(a,b){
    a=Math.max(0,Number(a)||0);b=Math.max(0,Number(b)||0);
    const aa=a*a,bb=b*b,d=aa+bb;
    return d>0?aa/d:1;
  }
});

const GGXImportanceSampler=Object.freeze({
  sampleHalfVector(n,roughness,u1,u2){
    const a=Math.max(.045,roughness);
    const a2=a*a;
    const phi=2*Math.PI*u1;
    const cosTheta=Math.sqrt(Math.max(0,(1-u2)/(1+(a2-1)*u2)));
    const sinTheta=Math.sqrt(Math.max(0,1-cosTheta*cosTheta));
    const basis=LightBasis.fromDirection(n);
    return RayMath.normalize(
      RayMath.add(
        RayMath.add(RayMath.mul(basis.tangent,sinTheta*Math.cos(phi)),RayMath.mul(basis.bitangent,sinTheta*Math.sin(phi))),
        RayMath.mul(basis.n,cosTheta)
      )
    );
  },
  reflect(i,n){return RayMath.sub(i,RayMath.mul(n,2*RayMath.dot(i,n)));},
  distribution(nDotH,roughness){
    const a=Math.max(.045,roughness),a2=a*a;
    const d=nDotH*nDotH*(a2-1)+1;
    return a2/(Math.PI*d*d+1e-8);
  }
});

const DielectricTransport=Object.freeze({
  fresnelDielectric(cosI,etaI,etaT){
    cosI=Math.max(-1,Math.min(1,cosI));
    let entering=cosI>0;
    if(!entering){const t=etaI;etaI=etaT;etaT=t;cosI=Math.abs(cosI);}
    const sinI=Math.sqrt(Math.max(0,1-cosI*cosI));
    const sinT=etaI/etaT*sinI;
    if(sinT>=1)return 1;
    const cosT=Math.sqrt(Math.max(0,1-sinT*sinT));

    const radiance=RenderColor.mul(tri.material.emissive,tri.material.emissiveIntensity);
    return {triangle:tri,point:lp,direction,distance,cosEmitter,pdf:pdfSolid,radiance};
  }
});

const LightTreePolicy=Object.freeze({
  threshold:6,
  enabled(job,compiled){
    return !!job?.settings?.lightTree && (compiled?.lights?.filter?.(l=>l.type!=='dome').length||0)>=this.threshold;
  },
  luminance(c){return .2126*c[0]+.7152*c[1]+.0722*c[2]},
  area(light){
    if(light.type==='rectangle'||light.type==='mesh')return Math.max(.01,light.width*light.height);
    if(light.type==='disc')return Math.max(.01,Math.PI*Math.pow(light.size*.5,2));
    if(light.type==='sphere')return Math.max(.01,4*Math.PI*Math.pow(light.size*.5,2));
    return 1;
  },
  power(light){
    const base=Math.max(1e-8,this.luminance(light.color||[1,1,1])*(light.intensity||0));
    return base*Math.sqrt(this.area(light));
  }
});
