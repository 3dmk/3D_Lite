var RenderGeometryPreparation={
  mainThread(job,packet){
    if(job.cancelled)throw Object.assign(new Error('Render job cancelled'),{cancelled:true});

    const vertexCount=packet.positions.length/3;
    const triangleCount=packet.indices.length/3;

    for(let i=0;i<packet.indices.length;i++){
      if(packet.indices[i]>=vertexCount)throw new GeometryCompileError('Index out of range in main-thread worker fallback');
    }

    return {
      type:'geometryPrepared',
      jobId:job.id,
      objectId:packet.objectId,
      vertexCount,
      triangleCount,
      bounds:new Float32Array([
        ...packet.bounds.min,
        ...packet.bounds.max
      ]),
      triangleIndices:new Uint32Array(packet.indices)
    };
  },

  async prepare(job,pool,onProgress=()=>{}){
    const packets=RenderGeometryCompiler.compileScene(job.renderScene);
    if(!packets.length)return new PreparedGeometrySet([]);

    const items=[];
    const total=packets.length;

    for(let i=0;i<packets.length;i++){
      if(job.cancelled)throw Object.assign(new Error('Render job cancelled'),{cancelled:true});

      while(job.paused&&!job.cancelled){
        await new Promise(resolve=>setTimeout(resolve,25));
      }

      const packet=packets[i];
      let result;

      try{
        result=await pool.run(
          {type:'prepareGeometry',jobId:job.id,packet},
          RenderGeometryCompiler.transferables(packet)
        );
      }catch(err){
        if(err?.workerUnavailable){
          result=this.mainThread(job,packet);
        }else{
          throw err;
        }
      }

      items.push(Object.freeze({

        bounds:Object.freeze(Array.from(result.bounds||[])),
        triangleIndices:result.triangleIndices

    for(let i=0;i<packet.positions.length;i++)this.positions.push(packet.positions[i]);

    // Startup-safe cross-generation compatibility.
    // Geometry packets created by earlier diagnostic/self-test paths may not
    // contain shading normals or UVs. Missing optional attributes must not
    // abort editor initialization.
    const packetVertexCount=packet.positions.length/3;
    const srcNormals=(packet.normals&&packet.normals.length===packet.positions.length)
      ? packet.normals
      : new Float32Array(packet.positions.length);
    const srcUVs=(packet.uvs&&packet.uvs.length===packetVertexCount*2)
      ? packet.uvs
      : new Float32Array(packetVertexCount*2);

    for(let i=0;i<srcNormals.length;i++)this.normals.push(srcNormals[i]);
    for(let i=0;i<srcUVs.length;i++)this.uvs.push(srcUVs[i]);
    const idx=packet.indices;
    for(let t=0;t<idx.length;t+=3){
      this.stats.inputTriangles++;
      const ia=idx[t],ib=idx[t+1],ic=idx[t+2];
      if(!Number.isInteger(ia)||!Number.isInteger(ib)||!Number.isInteger(ic)||ia<0||ib<0||ic<0||ia>=packetVertexCount||ib>=packetVertexCount||ic>=packetVertexCount){this.stats.rejectedInvalidIndex++;continue;}
      const ax=packet.positions[ia*3],ay=packet.positions[ia*3+1],az=packet.positions[ia*3+2];
      const bx=packet.positions[ib*3],by=packet.positions[ib*3+1],bz=packet.positions[ib*3+2];
      const cx=packet.positions[ic*3],cy=packet.positions[ic*3+1],cz=packet.positions[ic*3+2];
      if(!Number.isFinite(ax+ay+az+bx+by+bz+cx+cy+cz)){this.stats.rejectedNonFinite++;continue;}
      const abx=bx-ax,aby=by-ay,abz=bz-az,acx=cx-ax,acy=cy-ay,acz=cz-az;
      const nx=aby*acz-abz*acy,ny=abz*acx-abx*acz,nz=abx*acy-aby*acx;
      if(nx*nx+ny*ny+nz*nz<1e-20){this.stats.rejectedDegenerate++;continue;}
      this.triangles.push(baseVertex+ia,baseVertex+ib,baseVertex+ic);
      this.materialIndices.push(packet.materialIndices?.[t/3]||0);
      this.objectIndices.push(objectIndex);
      this.stats.acceptedTriangles++;
    }
    this.objectIds[objectIndex]=packet.objectId;
  }
  finalize(){
    this.positions=new Float32Array(this.positions);
    this.normals=new Float32Array(this.normals);
    this.uvs=new Float32Array(this.uvs);
    this.triangles=new Uint32Array(this.triangles);
    this.materialIndices=new Uint16Array(this.materialIndices);
    this.objectIndices=new Uint32Array(this.objectIndices);
    Object.freeze(this.objectIds);Object.freeze(this.stats);
    return this;
  }
  triangleCount(){return this.triangles.length/3}
  vertex(i){const p=i*3;return [this.positions[p],this.positions[p+1],this.positions[p+2]]}
  triangleBounds(t){
    const k=t*3,a=this.vertex(this.triangles[k]),b=this.vertex(this.triangles[k+1]),c=this.vertex(this.triangles[k+2]);
    return {min:RayMath.min3(a,RayMath.min3(b,c)),max:RayMath.max3(a,RayMath.max3(b,c)),
      centroid:[(a[0]+b[0]+c[0])/3,(a[1]+b[1]+c[1])/3,(a[2]+b[2]+c[2])/3]};
  }
}

class BVHNode{
  constructor(min,max,left=-1,right=-1,start=0,count=0){
    this.min=min;this.max=max;this.left=left;this.right=right;this.start=start;this.count=count;
  }
  get leaf(){return this.count>0}
}

var RenderBVH = class RenderBVH{
  constructor(store,maxLeafSize=4){
    this.store=store;
    this.maxLeafSize=Math.max(1,maxLeafSize|0);
    this.nodes=[];
    this.primitiveIndices=Uint32Array.from({length:store.triangleCount()},(_,i)=>i);
    this._boundsCache=Array.from({length:store.triangleCount()},(_,i)=>store.triangleBounds(i));
    if(store.triangleCount())this.root=this._build(0,store.triangleCount());
    else this.root=-1;
  }
  _rangeBounds(start,end){
    let mn=[Infinity,Infinity,Infinity],mx=[-Infinity,-Infinity,-Infinity],cmn=[Infinity,Infinity,Infinity],cmx=[-Infinity,-Infinity,-Infinity];
    for(let i=start;i<end;i++){
      const b=this._boundsCache[this.primitiveIndices[i]];
      mn=RayMath.min3(mn,b.min);mx=RayMath.max3(mx,b.max);
      cmn=RayMath.min3(cmn,b.centroid);cmx=RayMath.max3(cmx,b.centroid);
    }
    return {min:mn,max:mx,cmin:cmn,cmax:cmx};
  }
  _surfaceArea(min,max){
    const x=Math.max(0,max[0]-min[0]),y=Math.max(0,max[1]-min[1]),z=Math.max(0,max[2]-min[2]);
    return 2*(x*y+y*z+z*x);
  }
  _sahSplit(start,end,rb){
    const count=end-start;
    if(count<8)return null;
    let best=null,bestCost=Infinity;
    const binsN=8;
    for(let axis=0;axis<3;axis++){
      const lo=rb.cmin[axis],hi=rb.cmax[axis],span=hi-lo;
      if(!(span>1e-12))continue;
      const bins=Array.from({length:binsN},()=>({count:0,min:[Infinity,Infinity,Infinity],max:[-Infinity,-Infinity,-Infinity]}));
      for(let i=start;i<end;i++){
        const pi=this.primitiveIndices[i],b=this._boundsCache[pi];
        const bi=Math.min(binsN-1,Math.max(0,Math.floor(((b.centroid[axis]-lo)/span)*binsN)));
        const bn=bins[bi];bn.count++;bn.min=RayMath.min3(bn.min,b.min);bn.max=RayMath.max3(bn.max,b.max);
      }
      const leftC=[],rightC=[],leftB=[],rightB=[];
      let c=0,mn=[Infinity,Infinity,Infinity],mx=[-Infinity,-Infinity,-Infinity];
      for(let i=0;i<binsN;i++){c+=bins[i].count;if(bins[i].count){mn=RayMath.min3(mn,bins[i].min);mx=RayMath.max3(mx,bins[i].max);}leftC[i]=c;leftB[i]={min:mn.slice(),max:mx.slice()};}
      c=0;mn=[Infinity,Infinity,Infinity];mx=[-Infinity,-Infinity,-Infinity];
      for(let i=binsN-1;i>=0;i--){c+=bins[i].count;if(bins[i].count){mn=RayMath.min3(mn,bins[i].min);mx=RayMath.max3(mx,bins[i].max);}rightC[i]=c;rightB[i]={min:mn.slice(),max:mx.slice()};}
      for(let i=0;i<binsN-1;i++){
        if(!leftC[i]||!rightC[i+1])continue;
        const cost=this._surfaceArea(leftB[i].min,leftB[i].max)*leftC[i]+this._surfaceArea(rightB[i+1].min,rightB[i+1].max)*rightC[i+1];
        if(cost<bestCost){bestCost=cost;best={axis,threshold:lo+span*((i+1)/binsN)};}
      }
    }
    return best;
  }
  _build(start,end){
    const rb=this._rangeBounds(start,end);
    const nodeIndex=this.nodes.length;
    this.nodes.push(null);
    const count=end-start;
    if(count<=this.maxLeafSize){
      this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,-1,-1,start,count);
      return nodeIndex;
    }
    const ext=RayMath.sub(rb.cmax,rb.cmin);
    let axis=0;if(ext[1]>ext[axis])axis=1;if(ext[2]>ext[axis])axis=2;
    const sah=this._sahSplit(start,end,rb);
    if(sah)axis=sah.axis;
    const sorted=new Array(end-start);for(let si=0;si<sorted.length;si++)sorted[si]=this.primitiveIndices[start+si];
    sorted.sort((a,b)=>this._boundsCache[a].centroid[axis]-this._boundsCache[b].centroid[axis]);
    this.primitiveIndices.set(sorted,start);
    let mid=start+Math.floor(count/2);
    if(sah){
      let split=start;
      while(split<end&&this._boundsCache[this.primitiveIndices[split]].centroid[axis]<sah.threshold)split++;
      if(split>start&&split<end)mid=split;
    }
    if(mid===start||mid===end){
      this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,-1,-1,start,count);
      return nodeIndex;
    }
    const left=this._build(start,mid),right=this._build(mid,end);
    this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,left,right,0,0);
    return nodeIndex;
  }
  intersect(ray){
    const best=new RenderHit();
    if(this.root<0)return best;
    const stack=[this.root];stack.length=1; // traversal stack retained; avoids unsafe structural rewrite
    while(stack.length){
      const ni=stack.pop(),node=this.nodes[ni];
      if(!AABBIntersector.intersect(ray,node.min,node.max,best.t))continue;
      if(node.leaf){
        for(let i=node.start;i<node.start+node.count;i++){
          const tri=this.primitiveIndices[i],k=tri*3;
          const p0=this.store.vertex(this.store.triangles[k]);
          const p1=this.store.vertex(this.store.triangles[k+1]);
          const p2=this.store.vertex(this.store.triangles[k+2]);
          const before=best.t;
          TriangleIntersector.intersect(ray,p0,p1,p2,best);
          if(best.t<before){
            best.triangleIndex=tri;
            best.objectIndex=this.store.objectIndices[tri];
            best.materialIndex=this.store.materialIndices[tri];
          }
        }
      }else{
        const left=this.nodes[node.left],right=this.nodes[node.right];
        const lt=AABBIntersector.entry(ray,left.min,left.max,best.t);
        const rt=AABBIntersector.entry(ray,right.min,right.max,best.t);
        const lh=Number.isFinite(lt),rh=Number.isFinite(rt);
        if(lh&&rh){
          if(lt<=rt)stack.push(node.right,node.left);else stack.push(node.left,node.right);
        }else if(lh)stack.push(node.left);
        else if(rh)stack.push(node.right);
      }
    }
    return best;
  }
  stats(){
    let leafCount=0,maxDepth=0;
    if(this.root>=0){
      const stack=[[this.root,0]];
      while(stack.length){
        const [i,d]=stack.pop(),n=this.nodes[i];maxDepth=Math.max(maxDepth,d);
        if(n.leaf)leafCount++;else{stack.push([n.left,d+1]);stack.push([n.right,d+1]);}
      }
    }
    return Object.freeze({nodeCount:this.nodes.length,leafCount,maxDepth,triangleCount:this.store.triangleCount(),maxLeafSize:this.maxLeafSize,buildStrategy:'SAH-8bin-binary'});
  }
  anyHit(ray){
    if(this.root<0)return false;
    const stack=[this.root];
    const temp=new RenderHit();
    while(stack.length){
      const node=this.nodes[stack.pop()];
      if(!AABBIntersector.intersect(ray,node.min,node.max,ray.tMax))continue;
      if(node.leaf){
        for(let i=node.start;i<node.start+node.count;i++){
          const tri=this.primitiveIndices[i],k=tri*3;
          temp.hit=false;temp.t=Infinity;
          TriangleIntersector.intersect(
            ray,
            this.store.vertex(this.store.triangles[k]),
            this.store.vertex(this.store.triangles[k+1]),
            this.store.vertex(this.store.triangles[k+2]),
            temp
          );
          if(temp.hit)return true;
        }
      }else{
        stack.push(node.left,node.right);
      }
    }
    return false;
  }
}

const RenderCameraPolicy=Object.freeze({
  fallback:Object.freeze({
    type:'Perspective',
    position:Object.freeze([0,0,10]),
    quaternion:Object.freeze([0,0,0,1]),
    up:Object.freeze([0,0,1]),
    fov:50,aspect:1,near:1e-4,far:Infinity
  }),
  sanitize(snapshot){
    const s=snapshot||this.fallback;
    const position=(Array.isArray(s.position)&&s.position.length===3&&s.position.every(Number.isFinite))?s.position.slice():[...this.fallback.position];
    let q=(Array.isArray(s.quaternion)&&s.quaternion.length===4&&s.quaternion.every(Number.isFinite))?s.quaternion.slice():[0,0,0,1];
    const qLen=Math.hypot(q[0],q[1],q[2],q[3]);
    q=qLen>1e-12?q.map(v=>v/qLen):[0,0,0,1];
    const up=(Array.isArray(s.up)&&s.up.length===3&&s.up.every(Number.isFinite))?s.up.slice():[0,0,1];
    const fov=Math.max(1,Math.min(179,Number.isFinite(Number(s.fov))?Number(s.fov):50));
    const aspect=Math.max(1e-6,Number.isFinite(Number(s.aspect))?Number(s.aspect):1);
    const near=Math.max(RenderRayPolicy.epsilonForPoint(position),Number.isFinite(Number(s.near))?Number(s.near):1e-4);
    const far=(Number.isFinite(Number(s.far))&&Number(s.far)>near)?Number(s.far):Infinity;
    return {type:'Perspective',position,quaternion:q,up,fov,aspect,near,far};
  },
  outputAspect(width,height){
    width=Math.max(1,Number(width)||1);height=Math.max(1,Number(height)||1);
    return width/height;
  }
});

const RenderCameraRayGenerator=Object.freeze({
  rotateByQuaternion(v,q){
    const x=v[0],y=v[1],z=v[2],qx=q[0],qy=q[1],qz=q[2],qw=q[3];
    const ix=qw*x+qy*z-qz*y,iy=qw*y+qz*x-qx*z,iz=qw*z+qx*y-qy*x,iw=-qx*x-qy*y-qz*z;
    return [
      ix*qw+iw*-qx+iy*-qz-iz*-qy,
      iy*qw+iw*-qy+iz*-qx-ix*-qz,
      iz*qw+iw*-qz+ix*-qy-iy*-qx
    ];
  },
  ray(cameraSnapshot,px,py,width,height){
    const c=RenderCameraPolicy.sanitize(cameraSnapshot);
    width=Math.max(1,Number(width)||1);height=Math.max(1,Number(height)||1);
    const nx=((px+.5)/width)*2-1;
    const ny=1-((py+.5)/height)*2;
    const aspect=Math.max(1e-6,width/height);
    const fovRad=c.fov*Math.PI/180,tan=Math.tan(fovRad*.5);
    const local=RayMath.normalize([nx*aspect*tan,ny*tan,-1]);
    const world=RayMath.normalize(this.rotateByQuaternion(local,c.quaternion));
    return new RenderRay(c.position,world,c.near,c.far,RenderRayType.CAMERA,0);
  }
});

const RenderAccelerationCache16={
  geometryRevision:null,
  acceleration:null,
  hits:0,
  misses:0,
  reset(){this.geometryRevision=null;this.acceleration=null;this.hits=0;this.misses=0;},

  revision(scene){
    const m=scene?.metadata||{};
    return [m.sceneRevision??'s?',m.geometryRevision??'g?',m.materialRevision??'m?',m.lightRevision??'l?',scene?.objects?.length||0,
      ...(scene?.objects||[]).map(o=>`${o.id}:${o.visible!==false?1:0}:${o.mesh?.positions?.length||0}:${o.mesh?.indices?.length||0}`)
    ].join('|');
  },

    if(this.acceleration&&this.geometryRevision===rev){this.hits++;return this.acceleration;}

  },

    this.geometryRevision=this.revision(scene);
    this.acceleration=acceleration;
    return acceleration;
  },
  stats(){return Object.freeze({geometryRevision:this.geometryRevision,hits:this.hits,misses:this.misses,reused:!!this.acceleration});}
};

const AABBIntersector=Object.freeze({
  entry(ray,min,max,tLimit=Infinity){

        continue;

  constructor(){
    this.positions=[];
    this.normals=[];
    this.uvs=[];
    this.triangles=[];
    this.materialIndices=[];
    this.objectIndices=[];
    this.objectIds=[];
    this.stats={inputTriangles:0,acceptedTriangles:0,rejectedInvalidIndex:0,rejectedNonFinite:0,rejectedDegenerate:0};

      if(!Number.isFinite(ax+ay+az+bx+by+bz+cx+cy+cz)){this.stats.rejectedNonFinite++;continue;}
      const abx=bx-ax,aby=by-ay,abz=bz-az,acx=cx-ax,acy=cy-ay,acz=cz-az;
      const nx=aby*acz-abz*acy,ny=abz*acx-abx*acz,nz=abx*acy-aby*acx;
      if(nx*nx+ny*ny+nz*nz<1e-20){this.stats.rejectedDegenerate++;continue;}
      this.triangles.push(baseVertex+ia,baseVertex+ib,baseVertex+ic);
      this.materialIndices.push(packet.materialIndices?.[t/3]||0);
      this.objectIndices.push(objectIndex);
      this.stats.acceptedTriangles++;
    }
    this.objectIds[objectIndex]=packet.objectId;
  }
  finalize(){
    this.positions=new Float32Array(this.positions);
    this.normals=new Float32Array(this.normals);
    this.uvs=new Float32Array(this.uvs);
    this.triangles=new Uint32Array(this.triangles);
    this.materialIndices=new Uint16Array(this.materialIndices);
    this.objectIndices=new Uint32Array(this.objectIndices);
    Object.freeze(this.objectIds);Object.freeze(this.stats);
    return this;
  }
  triangleCount(){return this.triangles.length/3}
  vertex(i){const p=i*3;return [this.positions[p],this.positions[p+1],this.positions[p+2]]}
  triangleBounds(t){
    const k=t*3,a=this.vertex(this.triangles[k]),b=this.vertex(this.triangles[k+1]),c=this.vertex(this.triangles[k+2]);
    return {min:RayMath.min3(a,RayMath.min3(b,c)),max:RayMath.max3(a,RayMath.max3(b,c)),
      centroid:[(a[0]+b[0]+c[0])/3,(a[1]+b[1]+c[1])/3,(a[2]+b[2]+c[2])/3]};
  }
}

class BVHNode{
  constructor(min,max,left=-1,right=-1,start=0,count=0){
    this.min=min;this.max=max;this.left=left;this.right=right;this.start=start;this.count=count;
  }
  get leaf(){return this.count>0}

var RenderBVH = class RenderBVH{
  constructor(store,maxLeafSize=4){
    this.store=store;
    this.maxLeafSize=Math.max(1,maxLeafSize|0);
    this.nodes=[];
    this.primitiveIndices=Uint32Array.from({length:store.triangleCount()},(_,i)=>i);
    this._boundsCache=Array.from({length:store.triangleCount()},(_,i)=>store.triangleBounds(i));
    if(store.triangleCount())this.root=this._build(0,store.triangleCount());
    else this.root=-1;
  }
  _rangeBounds(start,end){
    let mn=[Infinity,Infinity,Infinity],mx=[-Infinity,-Infinity,-Infinity],cmn=[Infinity,Infinity,Infinity],cmx=[-Infinity,-Infinity,-Infinity];
    for(let i=start;i<end;i++){
      const b=this._boundsCache[this.primitiveIndices[i]];
      mn=RayMath.min3(mn,b.min);mx=RayMath.max3(mx,b.max);
      cmn=RayMath.min3(cmn,b.centroid);cmx=RayMath.max3(cmx,b.centroid);
    }
    return {min:mn,max:mx,cmin:cmn,cmax:cmx};
  }
  _surfaceArea(min,max){
    const x=Math.max(0,max[0]-min[0]),y=Math.max(0,max[1]-min[1]),z=Math.max(0,max[2]-min[2]);
    return 2*(x*y+y*z+z*x);
  }
  _sahSplit(start,end,rb){
    const count=end-start;
    if(count<8)return null;
    let best=null,bestCost=Infinity;
    const binsN=8;
    for(let axis=0;axis<3;axis++){
      const lo=rb.cmin[axis],hi=rb.cmax[axis],span=hi-lo;
      if(!(span>1e-12))continue;
      const bins=Array.from({length:binsN},()=>({count:0,min:[Infinity,Infinity,Infinity],max:[-Infinity,-Infinity,-Infinity]}));
      for(let i=start;i<end;i++){
        const pi=this.primitiveIndices[i],b=this._boundsCache[pi];
        const bi=Math.min(binsN-1,Math.max(0,Math.floor(((b.centroid[axis]-lo)/span)*binsN)));
        const bn=bins[bi];bn.count++;bn.min=RayMath.min3(bn.min,b.min);bn.max=RayMath.max3(bn.max,b.max);
      }
      const leftC=[],rightC=[],leftB=[],rightB=[];
      let c=0,mn=[Infinity,Infinity,Infinity],mx=[-Infinity,-Infinity,-Infinity];
      for(let i=0;i<binsN;i++){c+=bins[i].count;if(bins[i].count){mn=RayMath.min3(mn,bins[i].min);mx=RayMath.max3(mx,bins[i].max);}leftC[i]=c;leftB[i]={min:mn.slice(),max:mx.slice()};}
      c=0;mn=[Infinity,Infinity,Infinity];mx=[-Infinity,-Infinity,-Infinity];
      for(let i=binsN-1;i>=0;i--){c+=bins[i].count;if(bins[i].count){mn=RayMath.min3(mn,bins[i].min);mx=RayMath.max3(mx,bins[i].max);}rightC[i]=c;rightB[i]={min:mn.slice(),max:mx.slice()};}
      for(let i=0;i<binsN-1;i++){
        if(!leftC[i]||!rightC[i+1])continue;
        const cost=this._surfaceArea(leftB[i].min,leftB[i].max)*leftC[i]+this._surfaceArea(rightB[i+1].min,rightB[i+1].max)*rightC[i+1];
        if(cost<bestCost){bestCost=cost;best={axis,threshold:lo+span*((i+1)/binsN)};}
      }
    }
    return best;
  }
  _build(start,end){
    const rb=this._rangeBounds(start,end);
    const nodeIndex=this.nodes.length;
    this.nodes.push(null);
    const count=end-start;
    if(count<=this.maxLeafSize){
      this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,-1,-1,start,count);
      return nodeIndex;
    }
    const ext=RayMath.sub(rb.cmax,rb.cmin);
    let axis=0;if(ext[1]>ext[axis])axis=1;if(ext[2]>ext[axis])axis=2;
    const sah=this._sahSplit(start,end,rb);
    if(sah)axis=sah.axis;
    const sorted=new Array(end-start);for(let si=0;si<sorted.length;si++)sorted[si]=this.primitiveIndices[start+si];
    sorted.sort((a,b)=>this._boundsCache[a].centroid[axis]-this._boundsCache[b].centroid[axis]);
    this.primitiveIndices.set(sorted,start);
    let mid=start+Math.floor(count/2);
    if(sah){
      let split=start;
      while(split<end&&this._boundsCache[this.primitiveIndices[split]].centroid[axis]<sah.threshold)split++;
      if(split>start&&split<end)mid=split;
    }
    if(mid===start||mid===end){
      this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,-1,-1,start,count);
      return nodeIndex;
    }
    const left=this._build(start,mid),right=this._build(mid,end);
    this.nodes[nodeIndex]=new BVHNode(rb.min,rb.max,left,right,0,0);
    return nodeIndex;
  }
  intersect(ray){
    const best=new RenderHit();
    if(this.root<0)return best;
    const stack=[this.root];stack.length=1; // traversal stack retained; avoids unsafe structural rewrite
    while(stack.length){
      const ni=stack.pop(),node=this.nodes[ni];
      if(!AABBIntersector.intersect(ray,node.min,node.max,best.t))continue;
      if(node.leaf){
        for(let i=node.start;i<node.start+node.count;i++){
          const tri=this.primitiveIndices[i],k=tri*3;
          const p0=this.store.vertex(this.store.triangles[k]);
          const p1=this.store.vertex(this.store.triangles[k+1]);
          const p2=this.store.vertex(this.store.triangles[k+2]);
          const before=best.t;
          TriangleIntersector.intersect(ray,p0,p1,p2,best);
          if(best.t<before){
            best.triangleIndex=tri;
            best.objectIndex=this.store.objectIndices[tri];
            best.materialIndex=this.store.materialIndices[tri];
          }
        }
      }else{
        const left=this.nodes[node.left],right=this.nodes[node.right];
        const lt=AABBIntersector.entry(ray,left.min,left.max,best.t);
        const rt=AABBIntersector.entry(ray,right.min,right.max,best.t);
        const lh=Number.isFinite(lt),rh=Number.isFinite(rt);
        if(lh&&rh){
          if(lt<=rt)stack.push(node.right,node.left);else stack.push(node.left,node.right);
        }else if(lh)stack.push(node.left);
        else if(rh)stack.push(node.right);
      }
    }
    return best;
  }
  stats(){
    let leafCount=0,maxDepth=0;
    if(this.root>=0){
      const stack=[[this.root,0]];
      while(stack.length){
        const [i,d]=stack.pop(),n=this.nodes[i];maxDepth=Math.max(maxDepth,d);
        if(n.leaf)leafCount++;else{stack.push([n.left,d+1]);stack.push([n.right,d+1]);}
      }
    }
    return Object.freeze({nodeCount:this.nodes.length,leafCount,maxDepth,triangleCount:this.store.triangleCount(),maxLeafSize:this.maxLeafSize,buildStrategy:'SAH-8bin-binary'});
  }
  anyHit(ray){
    if(this.root<0)return false;
    const stack=[this.root];
    const temp=new RenderHit();
    while(stack.length){
      const node=this.nodes[stack.pop()];
      if(!AABBIntersector.intersect(ray,node.min,node.max,ray.tMax))continue;
      if(node.leaf){
        for(let i=node.start;i<node.start+node.count;i++){
          const tri=this.primitiveIndices[i],k=tri*3;
          temp.hit=false;temp.t=Infinity;
          TriangleIntersector.intersect(
            ray,
            this.store.vertex(this.store.triangles[k]),
            this.store.vertex(this.store.triangles[k+1]),
            this.store.vertex(this.store.triangles[k+2]),
            temp
          );
          if(temp.hit)return true;
        }
      }else{
        stack.push(node.left,node.right);
      }
    }
    return false;
  }
}

const RenderCameraPolicy=Object.freeze({
  fallback:Object.freeze({
    type:'Perspective',
    position:Object.freeze([0,0,10]),
    quaternion:Object.freeze([0,0,0,1]),
    up:Object.freeze([0,0,1]),
    fov:50,aspect:1,near:1e-4,far:Infinity
  }),
  sanitize(snapshot){
    const s=snapshot||this.fallback;
    const position=(Array.isArray(s.position)&&s.position.length===3&&s.position.every(Number.isFinite))?s.position.slice():[...this.fallback.position];
    let q=(Array.isArray(s.quaternion)&&s.quaternion.length===4&&s.quaternion.every(Number.isFinite))?s.quaternion.slice():[0,0,0,1];
    const qLen=Math.hypot(q[0],q[1],q[2],q[3]);
    q=qLen>1e-12?q.map(v=>v/qLen):[0,0,0,1];
    const up=(Array.isArray(s.up)&&s.up.length===3&&s.up.every(Number.isFinite))?s.up.slice():[0,0,1];
    const fov=Math.max(1,Math.min(179,Number.isFinite(Number(s.fov))?Number(s.fov):50));
    const aspect=Math.max(1e-6,Number.isFinite(Number(s.aspect))?Number(s.aspect):1);
    const near=Math.max(RenderRayPolicy.epsilonForPoint(position),Number.isFinite(Number(s.near))?Number(s.near):1e-4);
    const far=(Number.isFinite(Number(s.far))&&Number(s.far)>near)?Number(s.far):Infinity;
    return {type:'Perspective',position,quaternion:q,up,fov,aspect,near,far};
  },
  outputAspect(width,height){
    width=Math.max(1,Number(width)||1);height=Math.max(1,Number(height)||1);
    return width/height;
  }
});

const RenderCameraRayGenerator=Object.freeze({
  rotateByQuaternion(v,q){
    const x=v[0],y=v[1],z=v[2],qx=q[0],qy=q[1],qz=q[2],qw=q[3];
    const ix=qw*x+qy*z-qz*y,iy=qw*y+qz*x-qx*z,iz=qw*z+qx*y-qy*x,iw=-qx*x-qy*y-qz*z;
    return [
      ix*qw+iw*-qx+iy*-qz-iz*-qy,
      iy*qw+iw*-qy+iz*-qx-ix*-qz,
      iz*qw+iw*-qz+ix*-qy-iy*-qx
    ];
  },
  ray(cameraSnapshot,px,py,width,height){
    const c=RenderCameraPolicy.sanitize(cameraSnapshot);
    width=Math.max(1,Number(width)||1);height=Math.max(1,Number(height)||1);
    const nx=((px+.5)/width)*2-1;
    const ny=1-((py+.5)/height)*2;
    const aspect=Math.max(1e-6,width/height);
    const fovRad=c.fov*Math.PI/180,tan=Math.tan(fovRad*.5);
    const local=RayMath.normalize([nx*aspect*tan,ny*tan,-1]);
    const world=RayMath.normalize(this.rotateByQuaternion(local,c.quaternion));
    return new RenderRay(c.position,world,c.near,c.far,RenderRayType.CAMERA,0);
  }
});

const DiagnosticRenderer=Object.freeze({
  async render(job,acceleration,onProgress=()=>{}){
    const width=Math.min(640,Math.max(1,job.progress.resolution[0]));

    const rgba=new Uint8ClampedArray(width*height*4);
    const cameraSnapshot=job.renderScene.camera;
    const rowChunk=8;
    for(let y=0;y<height;y++){
      if(job.cancelled||job.controlToken16?.cancelled?.())throw Object.assign(new Error('Render job cancelled'),{cancelled:true});
      while((job.paused||job.controlToken16?.paused?.())&&!job.cancelled)await new Promise(r=>setTimeout(r,20));
      for(let x=0;x<width;x++){
        const ray=RenderCameraRayGenerator.ray(cameraSnapshot,x,y,width,height);
        perf16.addRay(RenderRayType.CAMERA);
        perf16.addRay(RenderRayType.CAMERA);
        const hit=acceleration.trace(ray);
        const o=(y*width+x)*4;
        if(hit.hit){
          const n=hit.geometricNormal;
          rgba[o]=Math.round((n[0]*.5+.5)*255);
          rgba[o+1]=Math.round((n[1]*.5+.5)*255);
          rgba[o+2]=Math.round((n[2]*.5+.5)*255);
          rgba[o+3]=255;
        }else{
          rgba[o]=24;rgba[o+1]=24;rgba[o+2]=24;rgba[o+3]=255;
        }
      }
      if(y%rowChunk===0)onProgress((y+1)/height);
      if(y%rowChunk===0)await new Promise(r=>setTimeout(r,0));
    }
    return {width,height,rgba};
  }
});

function validateRenderGroup4(){
  const errors=[];
  if(typeof RenderRay!=='function')errors.push('RenderRay missing');
  if(typeof TriangleIntersector.intersect!=='function')errors.push('Triangle intersector missing');
  if(typeof RenderBVH!=='function')errors.push('RenderBVH missing');
  if(typeof RenderCameraRayGenerator.ray!=='function')errors.push('Camera ray generator missing');
  if(typeof RenderAccelerationBuilder.build!=='function')errors.push('Acceleration builder missing');

  return {ok:errors.length===0,errors};
}

const RenderColor=Object.freeze({
  clamp01(x){return Math.max(0,Math.min(1,x))},
  add(a,b){return [a[0]+b[0],a[1]+b[1],a[2]+b[2]]},
  mul(a,b){
    if(Array.isArray(b))return [a[0]*b[0],a[1]*b[1],a[2]*b[2]];
    return [a[0]*b,a[1]*b,a[2]*b];
  },
  mix(a,b,t){return [a[0]+(b[0]-a[0])*t,a[1]+(b[1]-a[1])*t,a[2]+(b[2]-a[2])*t]},
  srgbToLinear1(x){
    x=this.clamp01(x);return x<=.04045?x/12.92:Math.pow((x+.055)/1.055,2.4);
  },
  linearToSrgb1(x){
    x=Math.max(0,x);return x<=.0031308?12.92*x:1.055*Math.pow(x,1/2.4)-.055;
  },
  srgbToLinear(c){return [this.srgbToLinear1(c[0]),this.srgbToLinear1(c[1]),this.srgbToLinear1(c[2])]},
  linearToSrgb(c){return [this.linearToSrgb1(c[0]),this.linearToSrgb1(c[1]),this.linearToSrgb1(c[2])]},
  hexToLinear(hex){
    const h=String(hex||'#202020').replace('#','');
    const n=parseInt(h.length===3?h.split('').map(x=>x+x).join(''):h,16);
    if(!Number.isFinite(n))return [0.0144,0.0144,0.0144];
    return this.srgbToLinear([((n>>16)&255)/255,((n>>8)&255)/255,(n&255)/255]);
  },
  aces(c){
    const f=x=>{
      const a=2.51,b=.03,d=2.43,e=.59,g=.14;
      return this.clamp01((x*(a*x+b))/(x*(d*x+e)+g));
    };
    return [f(c[0]),f(c[1]),f(c[2])];
  }
});

const RenderMaterialPolicy=Object.freeze({
  finiteColor(c,fallback=[.54,.54,.54]){
    if(!Array.isArray(c)||c.length<3)return fallback.slice();
    return c.slice(0,3).map((v,i)=>{
      v=Number(v);
      return Number.isFinite(v)?Math.max(0,Math.min(1,v)):fallback[i];
    });
  },
  scalar(v,fallback,min,max){
    v=Number(v);
    if(!Number.isFinite(v))v=fallback;
    return Math.max(min,Math.min(max,v));
  },
  shading(v){
    return ['physical','lambert','unlit'].includes(v)?v:'physical';
  }
});

class RenderSurfaceMaterial{
  constructor(snapshot){
    snapshot=snapshot||{};
    this.id=String(snapshot.id||'__material__');
    this.name=String(snapshot.name||'Material');
    this.shading=RenderMaterialPolicy.shading(snapshot.shading);
    const srgb=RenderMaterialPolicy.finiteColor(snapshot.baseColor,[.54,.54,.54]);
    this.baseColor=Object.freeze(RenderColor.srgbToLinear(srgb));
    this.roughness=RenderMaterialPolicy.scalar(snapshot.roughness,.65,.045,1);
    this.metalness=RenderMaterialPolicy.scalar(snapshot.metalness,0,0,1);
    this.opacity=RenderMaterialPolicy.scalar(snapshot.opacity,1,0,1);
    this.transparent=!!snapshot.transparent;
    this.transmission=RenderMaterialPolicy.scalar(snapshot.transmission,0,0,1);
    this.ior=RenderMaterialPolicy.scalar(snapshot.ior,1.5,1.0001,3);
    this.thickness=RenderMaterialPolicy.scalar(snapshot.thickness,0,0,1e9);
    const attenSrgb=RenderMaterialPolicy.finiteColor(snapshot.attenuationColor,[1,1,1]);
    this.attenuationColor=Object.freeze(RenderColor.srgbToLinear(attenSrgb));
    this.attenuationDistance=(Number.isFinite(Number(snapshot.attenuationDistance))&&Number(snapshot.attenuationDistance)>0)?Number(snapshot.attenuationDistance):Infinity;
    const emissiveSrgb=RenderMaterialPolicy.finiteColor(snapshot.emissive,[0,0,0]);
    this.emissive=Object.freeze(RenderColor.srgbToLinear(emissiveSrgb));
    this.emissiveIntensity=RenderMaterialPolicy.scalar(snapshot.emissiveIntensity,0,0,1e8);
    Object.freeze(this);
  }
}

  compile(renderScene){

      id:'__fallback__',name:'Fallback',baseColor:[.54,.54,.54],
      roughness:.65,metalness:0,opacity:1,emissive:[0,0,0],emissiveIntensity:0
    });
    return Object.freeze({
      list:Object.freeze(list),

      get(id){return byId.get(id)||fallback}

  },
  resolveHit(renderScene,library,hit){

  }

const PhysicalBSDF=Object.freeze({
  fresnelSchlick(cosTheta,f0){
    const m=Math.pow(1-Math.max(0,Math.min(1,cosTheta)),5);
    return [
      f0[0]+(1-f0[0])*m,
      f0[1]+(1-f0[1])*m,
      f0[2]+(1-f0[2])*m
    ];
  },
  distributionGGX(nDotH,roughness){
    const a=roughness*roughness,a2=a*a;
    const d=nDotH*nDotH*(a2-1)+1;
    return a2/Math.max(Math.PI*d*d,1e-7);
  },
  geometrySchlickGGX(nDotV,roughness){
    const r=roughness+1,k=(r*r)/8;
    return nDotV/Math.max(nDotV*(1-k)+k,1e-7);
  },
  geometrySmith(nDotV,nDotL,roughness){
    return this.geometrySchlickGGX(nDotV,roughness)*this.geometrySchlickGGX(nDotL,roughness);
  },
  evaluate(material,n,v,l){
    if(material.shading==='unlit')return material.baseColor;
    if(!material||![...n,...v,...l].every(Number.isFinite))return [0,0,0];

    const h=RayMath.normalize(RayMath.add(v,l));
    if(!h.every(Number.isFinite))return [0,0,0];
    const nDotL=Math.max(0,RayMath.dot(n,l));
    const nDotV=Math.max(0,RayMath.dot(n,v));
    const nDotH=Math.max(0,RayMath.dot(n,h));
    const vDotH=Math.max(0,RayMath.dot(v,h));
    if(nDotL<=0||nDotV<=0)return [0,0,0];

    const dielectricF0=[.04,.04,.04];
    const f0=RenderColor.mix(dielectricF0,material.baseColor,material.metalness);
    const F=this.fresnelSchlick(vDotH,f0);
    const D=this.distributionGGX(nDotH,material.roughness);
    const G=this.geometrySmith(nDotV,nDotL,material.roughness);

    const specScale=D*G/Math.max(4*nDotV*nDotL,1e-6);
    const spec=RenderColor.mul(F,specScale);

    const kD=[
      (1-F[0])*(1-material.metalness),
      (1-F[1])*(1-material.metalness),
      (1-F[2])*(1-material.metalness)
    ];
    const diffuse=[
      kD[0]*material.baseColor[0]/Math.PI,
      kD[1]*material.baseColor[1]/Math.PI,
      kD[2]*material.baseColor[2]/Math.PI
    ];
    const out=RenderColor.mul(RenderColor.add(diffuse,spec),nDotL);
    return out.map(x=>Number.isFinite(x)&&x>=0?x:0);
  }
});

function freezeRenderSnapshot(v){
  if(!v||typeof v!=='object'||Object.isFrozen(v)||ArrayBuffer.isView(v))return v;
  Object.freeze(v);for(const k of Object.keys(v))freezeRenderSnapshot(v[k]);return v;
}

const RenderRevisionPolicy=Object.freeze({
  keys:Object.freeze(['sceneRevision','geometryRevision','materialRevision','lightRevision','cameraRevision']),
  changed(previous,current){
    const p=previous?.metadata||{},c=current?.metadata||{};
    return {
      scene:p.sceneRevision!==c.sceneRevision,
      geometry:p.geometryRevision!==c.geometryRevision,
      material:p.materialRevision!==c.materialRevision,
      light:p.lightRevision!==c.lightRevision,
      camera:p.cameraRevision!==c.cameraRevision
    };
  },
  invalidation(previous,current){

    return Object.freeze({
      rebuildSnapshot:x.scene||x.geometry||x.material||x.light||x.camera,
      rebuildGeometry:x.scene||x.geometry,

const RenderSnapshotValidator={
  finite(a){if(!a||typeof a.length!=='number')return false;for(let i=0;i<a.length;i++)if(!Number.isFinite(a[i]))return false;return true},
  typed(a){return ArrayBuffer.isView(a)&&!(a instanceof DataView)},
  validate(scene){
    const errors=[];
    if(!(scene instanceof RenderSceneSnapshot)||!Object.isFrozen(scene))errors.push('Invalid immutable RenderSceneSnapshot');
    const cam=scene?.camera;
    if(!cam||!Array.isArray(cam.position)||cam.position.length!==3)errors.push('Invalid render camera position');
    if(cam&&(!Array.isArray(cam.quaternion)||cam.quaternion.length!==4))errors.push('Invalid render camera quaternion');
    if(cam&&(!this.finite(cam.position)||!this.finite(cam.quaternion||[])))errors.push('Non-finite render camera');
    if(cam&&(!Number.isFinite(cam.fov)||cam.fov<=0||cam.fov>=180))errors.push('Invalid render camera FOV');
    if(cam&&(!Number.isFinite(cam.near)||cam.near<0))errors.push('Invalid render camera near clip');
    if(cam&&Number.isFinite(cam.far)&&cam.far<=cam.near)errors.push('Invalid render camera far clip');

    const mids=new Set((scene?.materials||[]).map(m=>m.id)),oids=new Set();
    for(const o of scene?.objects||[]){
      if(oids.has(o.id))errors.push('Duplicate object id '+o.id);oids.add(o.id);
      const m=o.mesh;if(!m){errors.push('Missing mesh '+o.name);continue}
      if(!this.typed(m.positions)||!this.typed(m.indices))errors.push('Mesh buffers must be typed arrays '+o.name);
      if(m.positions.length%3)errors.push('Bad position buffer '+o.name);
      if(m.indices.length%3)errors.push('Bad triangle index buffer '+o.name);
      const vc=m.positions.length/3,tc=m.indices.length/3;
      if(m.normals?.length&&m.normals.length!==m.positions.length)errors.push('Normal mismatch '+o.name);
      if(m.uvs?.length&&m.uvs.length!==vc*2)errors.push('UV mismatch '+o.name);
      if(m.materialIndices?.length!==tc)errors.push('Material-index count mismatch '+o.name);
      if(!this.finite(m.positions)||!this.finite(m.normals||[])||!this.finite(m.uvs||[]))errors.push('Non-finite mesh '+o.name);
      for(let i=0;i<m.indices.length;i++){
        const id=m.indices[i];
        if(!Number.isInteger(id)||id<0||id>=vc){errors.push('Index out of range '+o.name);break}
      }
      const localMatCount=Math.max(1,o.materialIds?.length||0);
      for(let i=0;i<(m.materialIndices?.length||0);i++)if(m.materialIndices[i]>=localMatCount){errors.push('Material index out of range '+o.name);break}
      for(const id of o.materialIds||[])if(!mids.has(id))errors.push('Missing material '+id);
      const b=m.bounds;
      if(!b||!this.finite(b.min)||!this.finite(b.max)||b.min.length!==3||b.max.length!==3)errors.push('Invalid bounds '+o.name);
    }
    const md=scene?.metadata||{};
    for(const key of RenderRevisionPolicy.keys)if(!Number.isFinite(md[key]))errors.push('Missing revision '+key);
    return {ok:!errors.length,errors};
  }
};
var RenderSceneBoundary={
  v:new THREE.Vector3(),normalMatrix:new THREE.Matrix3(),
  materialId(m){return m?.uuid||String(m?.id??'default')},
  material(m){
    const c=m?.color;

      shading:m?.isMeshBasicMaterial?'unlit':m?.isMeshLambertMaterial?'lambert':'physical',
      baseColor:c?[c.r,c.g,c.b]:[.54,.54,.54],
      roughness:Number.isFinite(m?.roughness)?m.roughness:.65,
      metalness:Number.isFinite(m?.metalness)?m.metalness:0,
      opacity:Number.isFinite(m?.opacity)?m.opacity:1,

    });
  },
  bounds(p){
    let mn=[Infinity,Infinity,Infinity],mx=[-Infinity,-Infinity,-Infinity];
    for(let i=0;i<p.length;i+=3)for(let a=0;a<3;a++){const x=p[i+a];if(x<mn[a])mn[a]=x;if(x>mx[a])mx[a]=x}

  },
  mesh(obj,materialCount=1){

    const g=obj.geometry,pa=g?.getAttribute?.('position');if(!pa)return null;

    const p=new Float32Array(pa.count*3),n=new Float32Array(pa.count*3),uv=new Float32Array(pa.count*2);
    this.normalMatrix.getNormalMatrix(obj.matrixWorld);
    for(let i=0;i<pa.count;i++){
      this.v.fromBufferAttribute(pa,i).applyMatrix4(obj.matrixWorld);
      p[i*3]=this.v.x;p[i*3+1]=this.v.y;p[i*3+2]=this.v.z;
      if(na){this.v.fromBufferAttribute(na,i).applyMatrix3(this.normalMatrix).normalize();n[i*3]=this.v.x;n[i*3+1]=this.v.y;n[i*3+2]=this.v.z}
      if(ua){uv[i*2]=ua.getX(i);uv[i*2+1]=ua.getY(i)}
    }
    const idx=g.index?new Uint32Array(g.index.array):Uint32Array.from({length:pa.count},(_,i)=>i);
    const mi=new Uint16Array(Math.ceil(idx.length/3));
    for(const gr of g.groups||[]){
      const raw=Number(gr.materialIndex);
