function runRenderGroup5SelfTests(){
  const errors=[];
  const approx=(a,b,e=1e-4)=>Math.abs(a-b)<=e;

  const white=RenderColor.srgbToLinear([1,1,1]);
  if(!approx(white[0],1)||!approx(white[1],1)||!approx(white[2],1))errors.push('sRGB white conversion failed');

  const mat=new RenderSurfaceMaterial({
    id:'t',name:'t',shading:'physical',baseColor:[.8,.3,.2],
    roughness:.5,metalness:0,opacity:1,emissive:[0,0,0],emissiveIntensity:0
  });
  const brdf=PhysicalBSDF.evaluate(mat,[0,0,1],[0,0,1],[0,0,1]);
  if(brdf.some(x=>!Number.isFinite(x)||x<0))errors.push('Physical BRDF finite/positive test failed');

  const metallic=new RenderSurfaceMaterial({
    id:'m',name:'m',shading:'physical',baseColor:[.9,.7,.2],
    roughness:.25,metalness:1,opacity:1,emissive:[0,0,0],emissiveIntensity:0
  });
  const mb=PhysicalBSDF.evaluate(metallic,[0,0,1],[0,0,1],[0,0,1]);
  if(mb.some(x=>!Number.isFinite(x)||x<0))errors.push('Metallic BRDF test failed');

  const tm=RenderColor.aces([10,3,1]);
  if(tm.some(x=>x<0||x>1||!Number.isFinite(x)))errors.push('Tone mapping range test failed');

  return {ok:errors.length===0,errors};
}
window.runRenderGroup5SelfTests=runRenderGroup5SelfTests;

  const errors=[];

  return {ok:errors.length===0,errors};
}

var EditorLightSystem={
  nextId:1,
  isLightObject(obj){return !!obj?.userData?.renderLight},

  iconGeometry(type,size){
    const s=Math.max(2,Math.min(30,Number(size)||10));
    if(type==='rectangle')return new THREE.BoxGeometry(s*1.4,s*.08,s);
    if(type==='disc')return new THREE.CylinderGeometry(s*.55,s*.55,s*.08,16);
    if(type==='sphere'||type==='dome')return new THREE.SphereGeometry(s*.45,10,6);
    if(type==='spot')return new THREE.ConeGeometry(s*.35,s*.8,12);
    if(type==='sun')return new THREE.CylinderGeometry(s*.45,s*.45,s*.08,16);
    if(type==='mesh')return new THREE.BoxGeometry(s*.8,s*.8,s*.08);
    return new THREE.OctahedronGeometry(s*.4,0);
  },

  create(type,opts={}){
    type=LightTypes.includes(type)?type:'point';
    const d=LightDefaults[type]||LightDefaults.point;
    const size=Math.max(.01,Number(opts.size??d.size??25));
    const geometry=this.iconGeometry(type,size);
    const color=new THREE.Color(opts.color||'#fff2c8');
    const material=new THREE.MeshBasicMaterial({
      color,wireframe:true,transparent:true,opacity:.9,
      depthTest:true,depthWrite:false,toneMapped:false
    });
    const obj=new THREE.Mesh(geometry,material);
    obj.name=(type[0].toUpperCase()+type.slice(1))+' Light '+this.nextId++;
    obj.userData.renderLight={
      id:'light-'+Date.now().toString(36)+'-'+Math.random().toString(36).slice(2,7),
      type,
      enabled:opts.enabled!==false,
      color:'#'+color.getHexString(),
      intensity:Math.max(0,Number(opts.intensity??d.intensity??10)),
      range:Math.max(0,Number(opts.range??d.range??0)),
      size,
      width:Math.max(.01,Number(opts.width??d.width??size*2)),
      height:Math.max(.01,Number(opts.height??d.height??size*2)),
      coneAngle:Math.max(1,Math.min(179,Number(opts.coneAngle??d.coneAngle??45))),
      penumbra:Math.max(0,Math.min(1,Number(opts.penumbra??d.penumbra??.15))),
      castShadows:opts.castShadows!==false,
      renderOnly:true
    };
    obj.userData.isRenderLight=true;
    obj.position.set(0,0,Math.max(20,size));
    if(type==='sun'||type==='spot'||type==='rectangle'||type==='disc'||type==='mesh'){
      obj.rotation.x=-Math.PI*.25;
    }
    scene.add(obj);objects.push(obj);
    window.__3DLiteLightRevision=(window.__3DLiteLightRevision||0)+1;
    selectOnly(obj);
    refreshModifyPanel();
    requestRender();
    return obj;
  },

  updateIcon(obj){
    if(!this.isLightObject(obj))return;
    const l=obj.userData.renderLight;
    const old=obj.geometry;
    obj.geometry=this.iconGeometry(l.type,l.size);
    old?.dispose?.();
    if(obj.material?.color)obj.material.color.set(l.color||'#ffffff');
    requestRender();
  },

  serialize(obj){
    if(!this.isLightObject(obj))return null;
    obj.updateMatrixWorld(true);
    const l=obj.userData.renderLight;
    const q=obj.getWorldQuaternion(new THREE.Quaternion());
    const p=obj.getWorldPosition(new THREE.Vector3());
    const dir=new THREE.Vector3(0,0,-1).applyQuaternion(q).normalize();
    return {
      id:l.id||obj.uuid,
      name:obj.name||'Light',
      type:l.type,
      enabled:l.enabled!==false&&obj.visible!==false,
      color:l.color||'#ffffff',
      intensity:Math.max(0,Number(l.intensity)||0),
      range:Math.max(0,Number(l.range)||0),
      size:Math.max(.01,Number(l.size)||.01),
      width:Math.max(.01,Number(l.width)||.01),
      height:Math.max(.01,Number(l.height)||.01),
      coneAngle:Math.max(1,Math.min(179,Number(l.coneAngle)||45)),
      penumbra:Math.max(0,Math.min(1,Number(l.penumbra)||0)),
      castShadows:l.castShadows!==false,
      position:p.toArray(),
      quaternion:q.toArray(),
      direction:dir.toArray()
    };
  }
};
