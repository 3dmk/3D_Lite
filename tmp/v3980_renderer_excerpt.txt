function makeGrid(){
  const group=new THREE.Group();
  group.name='3DLiteGrid';

  const size=4000, step=100, half=size/2;
  const minor=[],major=[],axis=[];
  for(let i=-half;i<=half;i+=step){
    if(i===0)continue;
    const target=(Math.round(i/step)%5===0)?major:minor;
    target.push(i,-half,0, i,half,0);
    target.push(-half,i,0, half,i,0);
  }
  axis.push(-half,0,0, half,0,0);
  axis.push(0,-half,0, 0,half,0);

  // Slightly wider than one physical pixel. Screen-space width remains stable
  // when the camera is tilted, preventing jagged diagonal grid lines.
  const minorLine=cleanLineSegments(minor,cleanLineMaterial(COLORS.gridMinor,0.9,.76),'gridMinor');
  const majorLine=cleanLineSegments(major,cleanLineMaterial(COLORS.gridMajor,1.0,.90),'gridMajor');
  const axisLine=cleanLineSegments(axis,cleanLineMaterial(0x6d8190,1.12,.96),'gridAxis');
  minorLine.renderOrder=majorLine.renderOrder=axisLine.renderOrder=-10;
  group.add(minorLine,majorLine,axisLine);
  group.renderOrder=-10;
  group.matrixAutoUpdate=false;
  group.updateMatrix();
  scene.add(group);
}

  if(!mesh?.material)return [];
  return Array.isArray(mesh.material)?mesh.material:[mesh.material];
}

function applyBackSidesMaterial(mesh){
  const side=backSidesMaterialSide();
  for(const mat of objectMaterials(mesh)){
    if(!mat)continue;
    const resolvedSide=mat.userData?.forceDoubleSide ? THREE.DoubleSide : side;

    // Use Three.js native face culling only.
    // OFF = render front-facing triangles only.
    // ON  = render both front and back-facing triangles.
    if(mat.side!==resolvedSide){
      mat.side=resolvedSide;
      mat.needsUpdate=true;
    }
  }
}

const SurfaceInterpolator=Object.freeze({
  normal(acceleration,hit){
    const s=acceleration.store,t=hit.triangleIndex,k=t*3;
    if(!s.normals||s.normals.length!==s.positions.length)return hit.geometricNormal;
    const ia=s.triangles[k],ib=s.triangles[k+1],ic=s.triangles[k+2];
    const w=1-hit.u-hit.v;
    const read=i=>[s.normals[i*3],s.normals[i*3+1],s.normals[i*3+2]];
    const a=read(ia),b=read(ib),c=read(ic);
    let n=RayMath.normalize([
      a[0]*w+b[0]*hit.u+c[0]*hit.v,
      a[1]*w+b[1]*hit.u+c[1]*hit.v,
      a[2]*w+b[2]*hit.u+c[2]*hit.v
    ]);
    if(!n.every(Number.isFinite)||RayMath.length(n)<1e-10)n=hit.geometricNormal;
    if(RayMath.dot(n,hit.geometricNormal)<0)n=RayMath.mul(n,-1);
    return n;
  },
  uv(acceleration,hit){
    const s=acceleration.store,t=hit.triangleIndex,k=t*3;
    if(!s.uvs||s.uvs.length!==(s.positions.length/3)*2)return [0,0];
    const ia=s.triangles[k],ib=s.triangles[k+1],ic=s.triangles[k+2],w=1-hit.u-hit.v;
    return [
      s.uvs[ia*2]*w+s.uvs[ib*2]*hit.u+s.uvs[ic*2]*hit.v,
      s.uvs[ia*2+1]*w+s.uvs[ib*2+1]*hit.u+s.uvs[ic*2+1]*hit.v

  }

    const sample=EmissiveTriangleSet.sample(set,7,0,[0,0,2]);
    expect(sample&&sample.pdf>0&&sample.radiance.some(v=>v>0),'Emissive triangle sampling failed');
  }catch(e){errors.push('Emissive GI test: '+String(e?.message||e));}

  return {ok:errors.length===0,errors};
}
